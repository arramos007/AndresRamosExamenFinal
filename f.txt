qry05_MovimientosEconomicos


let
    Origen = qry01_RegistrosOriginales,

    AgregarClasificacion =
        Table.AddColumn(
            Origen,
            "CLASIFICACION_MOVIMIENTO",
            each
                let
                    NombreMovimiento =
                        if [#"NOMBRE MOV."] = null then
                            ""
                        else
                            Text.Upper(
                                Text.Trim(
                                    Text.From([#"NOMBRE MOV."])
                                )
                            ),

                    TipoMovimiento =
                        if [#"TIPO MOV."] = null then
                            ""
                        else
                            Text.Upper(
                                Text.Trim(
                                    Text.From([#"TIPO MOV."])
                                )
                            )
                in
                    if NombreMovimiento = "EMISION"
                        and TipoMovimiento = "CRE" then
                        "ACREDITACION_SRI"

                    else if NombreMovimiento = "ENDOSO"
                        and TipoMovimiento = "CRE" then
                        "ENDOSO_RECIBIDO"

                    else if NombreMovimiento = "ENDOSO"
                        and TipoMovimiento = "DEB" then
                        "ENDOSO_REALIZADO"

                    else if NombreMovimiento = "UTILIZACION" then
                        "UTILIZACION"

                    else
                        "OTRO_MOVIMIENTO_NO_CLASIFICADO",
            type text
        ),

    Resultado =
        Table.ReorderColumns(
            AgregarClasificacion,
            {"ID_REGISTRO_ORIGEN", "CLASIFICACION_MOVIMIENTO"}
                & List.RemoveItems(
                    Table.ColumnNames(AgregarClasificacion),
                    {
                        "ID_REGISTRO_ORIGEN",
                        "CLASIFICACION_MOVIMIENTO"
                    }
                )
        )
in
    Resultado
