qry31_ValidacionTotalesParticipantes



let
    Rutas = qry28_RutasEconomicasDirectas,
    Participantes = qry30_ResumenConexionesParticipantes,

    TotalOperaciones =
        Table.RowCount(Rutas),

    TotalRecibidoOrigen =
        List.Sum(Rutas[MONTO_DEBITO]),

    TotalRealizadoOrigen =
        List.Sum(Rutas[MONTO_DEBITO]),

    TotalRecibidoParticipantes =
        List.Sum(Participantes[TOTAL_RECIBIDO]),

    TotalRealizadoParticipantes =
        List.Sum(Participantes[TOTAL_REALIZADO]),

    CantidadRecibida =
        List.Sum(Participantes[CANTIDAD_ENDOSOS_RECIBIDOS]),

    CantidadRealizada =
        List.Sum(Participantes[CANTIDAD_ENDOSOS_REALIZADOS]),

    Validaciones = {
        {
            "TOTAL_OPERACIONES",
            TotalOperaciones,
            1338
        },
        {
            "CANTIDAD_ENDOSOS_RECIBIDOS",
            CantidadRecibida,
            TotalOperaciones
        },
        {
            "CANTIDAD_ENDOSOS_REALIZADOS",
            CantidadRealizada,
            TotalOperaciones
        },
        {
            "TOTAL_RECIBIDO",
            TotalRecibidoParticipantes,
            TotalRecibidoOrigen
        },
        {
            "TOTAL_REALIZADO",
            TotalRealizadoParticipantes,
            TotalRealizadoOrigen
        }
    },

    Tabla =
        Table.FromRows(
            Validaciones,
            {
                "VALIDACION",
                "RESULTADO",
                "ESPERADO"
            }
        ),

    AgregarDiferencia =
        Table.AddColumn(
            Tabla,
            "DIFERENCIA",
            each [RESULTADO] - [ESPERADO],
            type number
        ),

    AgregarEstado =
        Table.AddColumn(
            AgregarDiferencia,
            "ESTADO",
            each
                if Number.Abs([DIFERENCIA]) < 0.01 then
                    "CORRECTO"
                else
                    "REVISAR",
            type text
        )
in
    AgregarEstado
