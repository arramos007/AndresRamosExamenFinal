let
    Origen = qryDetalleDuplicado,

    Agrupar = Table.Group(
        Origen,
        {"RUC SOCIAL EMPLEADOR"},
        {
            {
                "TOTAL_REGISTROS",
                each Table.RowCount(_),
                Int64.Type
            },
            {
                "REGISTROS_DISTINTOS",
                each Table.RowCount(Table.Distinct(_)),
                Int64.Type
            },
            {
                "REPETICIONES_EXACTAS",
                each Table.RowCount(_) -
                     Table.RowCount(Table.Distinct(_)),
                Int64.Type
            },
            {
                "MONTO_UNICO",
                each List.First(
                    List.Distinct(Table.Column(_, "MONTO MOV."))
                ),
                type nullable number
            }
        }
    )
in
    Agrupar
