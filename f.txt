fnExpandirUnNivel


(Estado as table, Indice as table) as list =>
let
    // Procesar cada ruta que todavía está abierta
    Rutas =
        Table.ToRecords(
            Table.SelectRows(
                Estado,
                each [ESTADO_RUTA] = "ABIERTA"
            )
        ),

    // Buscar las conexiones posteriores de cada ruta
    ExpandirRutas =
        List.Transform(
            Rutas,
            (Padre) =>
                let
                    ParticipanteActual =
                        Padre[ID_PARTICIPANTE_ACTUAL],

                    FechaAnterior =
                        Padre[FECHA_ULTIMO_MOVIMIENTO],

                    OperacionesAnteriores =
                        Padre[OPERACIONES_RECORRIDAS],

                    ParticipantesAnteriores =
                        Padre[PARTICIPANTES_VISITADOS],

                    // Localizar salidas del participante actual
                    BuscarIndice =
                        Table.SelectRows(
                            Indice,
                            each
                                [ID_PARTICIPANTE_ORIGEN]
                                = ParticipanteActual
                        ),

                    Salidas =
                        if Table.IsEmpty(BuscarIndice) then
                            #table({}, {})
                        else
                            BuscarIndice{0}[CONEXIONES_SALIDA],

                    // Evitar operaciones anteriores y fechas previas
                    Candidatos =
                        Table.SelectRows(
                            Salidas,
                            each
                                [FECHA_DEBITO] >= FechaAnterior
                                and not List.Contains(
                                    OperacionesAnteriores,
                                    [ID_REGISTRO_DEBITO]
                                )
                        ),

                    Registros =
                        Table.ToRecords(Candidatos),

                    // Construir cada nueva rama
                    NuevasRutas =
                        List.Transform(
                            Registros,
                            (Conexion) =>
                                let
                                    Destino =
                                        Conexion[ID_PARTICIPANTE_DESTINO],

                                    FechaIgual =
                                        Conexion[FECHA_DEBITO]
                                        = FechaAnterior,

                                    Ciclo =
                                        List.Contains(
                                            ParticipantesAnteriores,
                                            Destino
                                        ),

                                    EstadoNuevaRuta =
                                        if Ciclo then
                                            "CICLO_DETECTADO"
                                        else if FechaIgual then
                                            "ORDEN_TEMPORAL_AMBIGUO"
                                        else
                                            "ABIERTA",

                                    Evidencia =
                                        if Ciclo or FechaIgual then
                                            "CONEXION_OBSERVADA"
                                        else if
                                            Padre[NIVEL_EVIDENCIA_RUTA]
                                            = "CONFIRMADA"
                                            and Conexion[NIVEL_EVIDENCIA]
                                            = "CONFIRMADA"
                                        then
                                            "CONFIRMADA"
                                        else
                                            "PARCIALMENTE_TRAZABLE",

                                    NuevaRuta =
                                        [
                                            ID_RUTA_ACTUAL =
                                                Padre[ID_RUTA_ACTUAL]
                                                & ">"
                                                & Text.From(
                                                    Conexion[ID_REGISTRO_DEBITO]
                                                ),

                                            ID_RUTA_PADRE =
                                                Padre[ID_RUTA_ACTUAL],

                                            ID_REGISTRO_DEBITO =
                                                Conexion[ID_REGISTRO_DEBITO],

                                            ID_REGISTRO_CREDITO =
                                                Conexion[ID_REGISTRO_CREDITO],

                                            ID_PARTICIPANTE_INICIAL =
                                                Padre[ID_PARTICIPANTE_INICIAL],

                                            ID_PARTICIPANTE_ACTUAL =
                                                Destino,

                                            NUMERO_SALTO =
                                                Padre[NUMERO_SALTO] + 1,

                                            PARTICIPANTES_VISITADOS =
                                                ParticipantesAnteriores
                                                & {Destino},

                                            OPERACIONES_RECORRIDAS =
                                                OperacionesAnteriores
                                                & {
                                                    Conexion[ID_REGISTRO_DEBITO]
                                                },

                                            FECHA_DEBITO =
                                                Conexion[FECHA_DEBITO],

                                            FECHA_CREDITO =
                                                Conexion[FECHA_CREDITO],

                                            FECHA_ULTIMO_MOVIMIENTO =
                                                Conexion[FECHA_CREDITO],

                                            MONTO_DEBITO =
                                                Conexion[MONTO_DEBITO],

                                            MONTO_CREDITO =
                                                Conexion[MONTO_CREDITO],

                                            ESTADO_RUTA =
                                                EstadoNuevaRuta,

                                            ES_CICLO =
                                                Ciclo,

                                            NIVEL_EVIDENCIA_RUTA =
                                                Evidencia
                                        ]
                                in
                                    NuevaRuta
                        )
                in
                    NuevasRutas
        ),

    Resultado =
        List.Combine(ExpandirRutas)
in
    Resultado
