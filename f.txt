qry20_RankingEntidadesReceptoras


let
    Origen = qry17_EndososConRelacionesEmpresariales,

    // 1. Conservar entidades receptoras con RUC de 13 dígitos
    FiltrarRUC =
        Table.SelectRows(
            Origen,
            each
                [RUC_EMPRESA_BENEFICIARIO] <> null
                and Text.Length(
                    Text.Trim(
                        Text.From([RUC_EMPRESA_BENEFICIARIO])
                    )
                ) = 13
        ),

    // 2. Clasificar por tercer dígito
    AgregarTipoEntidad =
        Table.AddColumn(
            FiltrarRUC,
            "TIPO_ENTIDAD",
            each
                let
                    Digito =
                        Text.Middle(
                            Text.Trim(
                                Text.From([RUC_EMPRESA_BENEFICIARIO])
                            ),
                            2,
                            1
                        )
                in
                    if Digito = "9" then
                        "SOCIEDAD_PRIVADA"
                    else if Digito = "6" then
                        "ENTIDAD_PUBLICA"
                    else
                        "PERSONA_NATURAL",
            type text
        ),

    // 3. Incluir sociedades privadas y entidades públicas
    FiltrarEntidades =
        Table.SelectRows(
            AgregarTipoEntidad,
            each
                [TIPO_ENTIDAD] = "SOCIEDAD_PRIVADA"
                or [TIPO_ENTIDAD] = "ENTIDAD_PUBLICA"
        ),

    // 4. Agrupar por entidad receptora
    AgruparEntidades =
        Table.Group(
            FiltrarEntidades,
            {
                "RUC_EMPRESA_BENEFICIARIO",
                "RAZON_SOCIAL_BENEFICIARIO",
                "TIPO_ENTIDAD"
            },
            {
                {
                    "CANTIDAD_ENDOSOS_RECIBIDOS",
                    each
                        List.Count(
                            List.Distinct(
                                List.RemoveNulls(
                                    [ID_OPERACION]
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "IMPORTE_REFERENCIAL_RECIBIDO",
                    each
                        let
                            Operaciones =
                                Table.Distinct(
                                    Table.SelectColumns(
                                        _,
                                        {
                                            "ID_OPERACION",
                                            "MONTO_DEBITO"
                                        }
                                    ),
                                    {"ID_OPERACION"}
                                )
                        in
                            List.Sum(
                                List.RemoveNulls(
                                    Operaciones[MONTO_DEBITO]
                                )
                            ),
                    type number
                },
                {
                    "CANTIDAD_EMPRESAS_EMISORAS",
                    each
                        List.Count(
                            List.Distinct(
                                List.RemoveNulls(
                                    [RUC_EMPRESA_EMISOR]
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "CANTIDAD_PARTICIPANTES_EMISORES",
                    each
                        List.Count(
                            List.Distinct(
                                List.RemoveNulls(
                                    [ID_PARTICIPANTE_EMISOR_CARTOLA]
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "PRIMERA_FECHA_RECEPCION",
                    each List.Min(
                        List.RemoveNulls([FECHA_DEBITO])
                    ),
                    type nullable datetime
                },
                {
                    "ULTIMA_FECHA_RECEPCION",
                    each List.Max(
                        List.RemoveNulls([FECHA_DEBITO])
                    ),
                    type nullable datetime
                }
            }
        ),

    // 5. Ordenar por importe referencial
    OrdenarRanking =
        Table.Sort(
            AgruparEntidades,
            {
                {"IMPORTE_REFERENCIAL_RECIBIDO", Order.Descending},
                {"CANTIDAD_ENDOSOS_RECIBIDOS", Order.Descending}
            }
        ),

    // 6. Agregar posición del ranking
    AgregarRanking =
        Table.AddIndexColumn(
            OrdenarRanking,
            "POSICION_RANKING",
            1,
            1,
            Int64.Type
        ),

    Resultado =
        Table.ReorderColumns(
            AgregarRanking,
            {
                "POSICION_RANKING",
                "RUC_EMPRESA_BENEFICIARIO",
                "RAZON_SOCIAL_BENEFICIARIO",
                "TIPO_ENTIDAD",
                "CANTIDAD_ENDOSOS_RECIBIDOS",
                "IMPORTE_REFERENCIAL_RECIBIDO",
                "CANTIDAD_EMPRESAS_EMISORAS",
                "CANTIDAD_PARTICIPANTES_EMISORES",
                "PRIMERA_FECHA_RECEPCION",
                "ULTIMA_FECHA_RECEPCION"
            }
        )
in
    Resultado
