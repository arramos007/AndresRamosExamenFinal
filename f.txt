let
    Origen = qryModeloMovimientos,

    FiltrarEndosos = Table.SelectRows(
        Origen,
        each
            [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO"
            or [CLASIFICACION_MOVIMIENTO] = "ENDOSO_RECIBIDO"
    ),

    SeleccionarColumnas = Table.SelectColumns(
        FiltrarEndosos,
        {
            "REL",
            "RELAC.",
            "RUC SOCIAL EMPLEADOR",
            "RAZON SOCIAL EMPLEADOR",
            "IDENTIFICACION EMPLEADO (EF)",
            "NOMBRE EMPLEADO (EF)",
            "ID_TITULAR",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "FECHA DE MOV.",
            "CODIGO MOV.",
            "NOMBRE MOV.",
            "TIPO MOV.",
            "TIPO_FLUJO",
            "TIPO_OPERACION",
            "MONTO MOV.",
            "SALDO NCD",
            "EN TRAMA EMISOR",
            "EN TRAMA BENEF.",
            "CLAVE_MOVIMIENTO",
            "ESTADO_TRAZABILIDAD"
        }
    ),

    ValidarTitularCartola = Table.AddColumn(
        SeleccionarColumnas,
        "VALIDACION_LADO_CARTOLA",
        each
            let
                Tipo = Text.Upper(Text.Trim(
                    if [#"TIPO MOV."] = null then ""
                    else [#"TIPO MOV."]
                )),
                Titular = [ID_TITULAR],
                Emisor = [ID_EMISOR],
                Beneficiario = [ID_BENEFICIARIO]
            in
                if Tipo = "DEB" then
                    if Titular = Emisor then "DEBITO_VALIDO"
                    else "DEBITO_NO_COINCIDE_CON_EMISOR"
                else if Tipo = "CRE" then
                    if Titular = Beneficiario then "CREDITO_VALIDO"
                    else "CREDITO_NO_COINCIDE_CON_BENEFICIARIO"
                else "TIPO_MOVIMIENTO_NO_RECONOCIDO",
        type text
    ),

    IdentificarContraparte = Table.AddColumn(
        ValidarTitularCartola,
        "ID_CONTRAPARTE_ENDOSO",
        each
            if [VALIDACION_LADO_CARTOLA] = "DEBITO_VALIDO" then
                [ID_BENEFICIARIO]
            else if [VALIDACION_LADO_CARTOLA] = "CREDITO_VALIDO" then
                [ID_EMISOR]
            else null,
        type text
    ),

    OrdenarPorFecha = Table.Sort(
        IdentificarContraparte,
        {{"FECHA DE MOV.", Order.Ascending}}
    )
in
    OrdenarPorFecha
