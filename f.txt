= Table.AddColumn(
    #"Se expandió PERFIL_TRANSACCIONAL",
    "PERFIL_RED",
    each
        let
            Recibidos = [TOTAL_RECIBIDOS_TRAMA],
            Salidas = [TOTAL_SALIDAS_TRAMA],

            Perfiles = List.RemoveNulls({
                if Recibidos > 0 then "CONCENTRADOR" else null,
                if Recibidos > 0 and Salidas > 0 then "ACUMULADOR" else null
            })
        in
            if List.Count(Perfiles) = 0
            then "SIN PERFIL IDENTIFICADO"
            else Text.Combine(Perfiles, " | "),
    type text
)
