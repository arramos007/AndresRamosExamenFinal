let
    Origen = qryAuditoriaDuplicadosExactos,

    Agrupar = Table.Group(
        Origen,
        {
            "IDENTIFICACION EMPLEADO (EF)",
            "RUC SOCIAL EMPLEADOR"
        },
        {
            {
                "CODIGOS_AFECTADOS",
                each List.Count(
                    List.Distinct(
                        Table.Column(_, "CODIGO MOV.")
                    )
                ),
                Int64.Type
            },
            {
                "REGISTROS_EXCEDENTES",
                each List.Sum(
                    List.RemoveNulls(
                        Table.Column(_, "REGISTROS_EXCEDENTES")
                    )
                ),
                Int64.Type
            },
            {
                "MONTO_EXCEDENTE_CANDIDATO",
                each List.Sum(
                    List.RemoveNulls(
                        Table.Column(_, "MONTO_EXCEDENTE_CANDIDATO")
                    )
                ),
                type number
            }
        }
    ),

    Ordenar = Table.Sort(
        Agrupar,
        {
            {"MONTO_EXCEDENTE_CANDIDATO", Order.Descending}
        }
    )
in
    Ordenar
