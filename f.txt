qry12_ClasificacionCandidatos


let
    Origen = qry11_CandidatosConciliacion,

    FiltrarCandidatos =
        Table.SelectRows(
            Origen,
            each [TIPO_REGISTRO] = "CANDIDATO"
        ),

    // Contar cuántos débitos apuntan a cada crédito
    AgruparPorCredito =
        Table.Group(
            FiltrarCandidatos,
            {"ID_REGISTRO_CREDITO"},
            {
                {
                    "CANTIDAD_CANDIDATOS_CREDITO",
                    each Table.RowCount(_),
                    Int64.Type
                }
            }
        ),

    UnirConteoCredito =
        Table.NestedJoin(
            FiltrarCandidatos,
            {"ID_REGISTRO_CREDITO"},
            AgruparPorCredito,
            {"ID_REGISTRO_CREDITO"},
            "CONTEO_CREDITO",
            JoinKind.LeftOuter
        ),

    ExpandirConteoCredito =
        Table.ExpandTableColumn(
            UnirConteoCredito,
            "CONTEO_CREDITO",
            {"CANTIDAD_CANDIDATOS_CREDITO"},
            {"CANTIDAD_CANDIDATOS_CREDITO"}
        ),

    // Clasificar considerando ambos extremos
    AgregarClasificacion =
        Table.AddColumn(
            ExpandirConteoCredito,
            "ESTADO_CANDIDATO",
            each
                if
                    [CANTIDAD_CANDIDATOS_DEBITO] = 1
                    and [CANTIDAD_CANDIDATOS_CREDITO] = 1
                then
                    "CANDIDATO_UNICO"
                else
                    "AMBIGUO",
            type text
        ),

    Resultado =
        Table.Sort(
            AgregarClasificacion,
            {
                {"ID_REGISTRO_DEBITO", Order.Ascending},
                {"ID_REGISTRO_CREDITO", Order.Ascending}
            }
        )
in
    Resultado
