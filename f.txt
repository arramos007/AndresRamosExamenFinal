
let
    Origen = qryDetalleEndosos,

    // =====================================================
    // 1. PARAMETRO TEMPORAL
    // =====================================================
    ToleranciaMinutos = 10,

    // =====================================================
    // 2. SEPARAR DEBITOS Y CREDITOS
    // =====================================================
    DebitosOrigen = Table.SelectRows(
        Origen,
        each
            [TIPO_CARTOLA] = "DEB"
            and [ID_EMISOR_CARTOLA] <> null
            and [ID_CONTRAPARTE_ENDOSO] <> null
            and [#"MONTO MOV."] <> null
            and [#"FECHA DE MOV."] <> null
    ),

    CreditosOrigen = Table.SelectRows(
        Origen,
        each
            [TIPO_CARTOLA] = "CRE"
            and [ID_BENEFICIARIO_CARTOLA] <> null
            and [#"MONTO MOV."] <> null
            and [#"FECHA DE MOV."] <> null
    ),

    // =====================================================
    // 3. ELIMINAR DUPLICADOS ADMINISTRATIVOS
    // =====================================================
    DebitosUnicos = Table.Distinct(
        DebitosOrigen,
        {
            "ID_EMISOR_CARTOLA",
            "ID_CONTRAPARTE_ENDOSO",
            "MONTO MOV.",
            "FECHA DE MOV.",
            "CODIGO MOV."
        }
    ),

    CreditosUnicos = Table.Distinct(
        CreditosOrigen,
        {
            "ID_BENEFICIARIO_CARTOLA",
            "MONTO MOV.",
            "FECHA DE MOV.",
            "CODIGO MOV."
        }
    ),

    // =====================================================
    // 4. IDENTIFICAR CADA REGISTRO
    // =====================================================
    DebitosIndexados = Table.AddIndexColumn(
        DebitosUnicos,
        "ID_DEBITO",
        1,
        1,
        Int64.Type
    ),

    CreditosIndexados = Table.AddIndexColumn(
        CreditosUnicos,
        "ID_CREDITO",
        1,
        1,
        Int64.Type
    ),

    // =====================================================
    // 5. CRUCE OPTIMIZADO POR BENEFICIARIO E IMPORTE
    // =====================================================
    CruceInicial = Table.NestedJoin(
        DebitosIndexados,
        {
            "ID_CONTRAPARTE_ENDOSO",
            "MONTO MOV."
        },
        CreditosIndexados,
        {
            "ID_BENEFICIARIO_CARTOLA",
            "MONTO MOV."
        },
        "CANDIDATOS",
        JoinKind.Inner
    ),

    ExpandirCandidatos = Table.ExpandTableColumn(
        CruceInicial,
        "CANDIDATOS",
        {
            "ID_CREDITO",
            "FECHA DE MOV.",
            "CODIGO MOV."
        },
        {
            "ID_CREDITO",
            "FECHA_CREDITO",
            "CODIGO_CREDITO"
        }
    ),

    // =====================================================
    // 6. APLICAR VENTANA TEMPORAL
    // =====================================================
    FiltrarTiempo = Table.SelectRows(
        ExpandirCandidatos,
        each
            Number.Abs(
                Duration.TotalMinutes(
                    [FECHA_CREDITO] - [#"FECHA DE MOV."]
                )
            ) <= ToleranciaMinutos
    ),

    AgregarFechaDebito = Table.AddColumn(
        FiltrarTiempo,
        "FECHA_DEBITO",
        each [#"FECHA DE MOV."],
        type datetime
    ),

    AgregarCodigoDebito = Table.AddColumn(
        AgregarFechaDebito,
        "CODIGO_DEBITO",
        each [#"CODIGO MOV."]
    ),

    AgregarDiferencia = Table.AddColumn(
        AgregarCodigoDebito,
        "DIFERENCIA_MINUTOS",
        each
            Number.Abs(
                Duration.TotalMinutes(
                    [FECHA_CREDITO] - [FECHA_DEBITO]
                )
            ),
        type number
    ),

    // =====================================================
    // 7. CONTAR CANDIDATOS POR DEBITO
    // =====================================================
    ConteoDebitos = Table.Group(
        AgregarDiferencia,
        {"ID_DEBITO"},
        {
            {
                "TOTAL_CANDIDATOS_DEBITO",
                each Table.RowCount(_),
                Int64.Type
            }
        }
    ),

    // =====================================================
    // 8. CONTAR CANDIDATOS POR CREDITO
    // =====================================================
    ConteoCreditos = Table.Group(
        AgregarDiferencia,
        {"ID_CREDITO"},
        {
            {
                "TOTAL_CANDIDATOS_CREDITO",
                each Table.RowCount(_),
                Int64.Type
            }
        }
    ),

    // =====================================================
    // 9. AGREGAR AMBOS CONTADORES A LAS POSIBLES PAREJAS
    // =====================================================
    UnirConteoDebitos = Table.NestedJoin(
        AgregarDiferencia,
        {"ID_DEBITO"},
        ConteoDebitos,
        {"ID_DEBITO"},
        "CONTEO_DEBITOS",
        JoinKind.LeftOuter
    ),

    ExpandirConteoDebitos = Table.ExpandTableColumn(
        UnirConteoDebitos,
        "CONTEO_DEBITOS",
        {"TOTAL_CANDIDATOS_DEBITO"},
        {"TOTAL_CANDIDATOS_DEBITO"}
    ),

    UnirConteoCreditos = Table.NestedJoin(
        ExpandirConteoDebitos,
        {"ID_CREDITO"},
        ConteoCreditos,
        {"ID_CREDITO"},
        "CONTEO_CREDITOS",
        JoinKind.LeftOuter
    ),

    ExpandirConteoCreditos = Table.ExpandTableColumn(
        UnirConteoCreditos,
        "CONTEO_CREDITOS",
        {"TOTAL_CANDIDATOS_CREDITO"},
        {"TOTAL_CANDIDATOS_CREDITO"}
    ),

    // =====================================================
    // 10. CONFIRMAR SOLAMENTE PAREJAS MUTUAMENTE UNICAS
    // =====================================================
    ParejasConfirmadas = Table.SelectRows(
        ExpandirConteoCreditos,
        each
            [TOTAL_CANDIDATOS_DEBITO] = 1
            and [TOTAL_CANDIDATOS_CREDITO] = 1
    ),

    CreditosConfirmados = List.Distinct(
        Table.Column(
            ParejasConfirmadas,
            "ID_CREDITO"
        )
    ),

    // =====================================================
    // 11. RESULTADO DE LOS DEBITOS
    // =====================================================
    UnirConteoDebitoOriginal = Table.NestedJoin(
        DebitosIndexados,
        {"ID_DEBITO"},
        ConteoDebitos,
        {"ID_DEBITO"},
        "CONTEO",
        JoinKind.LeftOuter
    ),

    ExpandirConteoDebitoOriginal = Table.ExpandTableColumn(
        UnirConteoDebitoOriginal,
        "CONTEO",
        {"TOTAL_CANDIDATOS_DEBITO"},
        {"TOTAL_CANDIDATOS_DEBITO"}
    ),

    ReemplazarConteoNuloDebito = Table.ReplaceValue(
        ExpandirConteoDebitoOriginal,
        null,
        0,
        Replacer.ReplaceValue,
        {"TOTAL_CANDIDATOS_DEBITO"}
    ),

    UnirParejasConfirmadas = Table.NestedJoin(
        ReemplazarConteoNuloDebito,
        {"ID_DEBITO"},
        ParejasConfirmadas,
        {"ID_DEBITO"},
        "PAREJA",
        JoinKind.LeftOuter
    ),

    ExpandirParejasConfirmadas = Table.ExpandTableColumn(
        UnirParejasConfirmadas,
        "PAREJA",
        {
            "ID_CREDITO",
            "FECHA_CREDITO",
            "CODIGO_CREDITO",
            "DIFERENCIA_MINUTOS",
            "TOTAL_CANDIDATOS_CREDITO"
        },
        {
            "ID_CREDITO",
            "FECHA_CREDITO",
            "CODIGO_CREDITO",
            "DIFERENCIA_MINUTOS",
            "TOTAL_CANDIDATOS_CREDITO"
        }
    ),

    ClasificarDebitos = Table.AddColumn(
        ExpandirParejasConfirmadas,
        "ESTADO_CONCILIACION",
        each
            if [TOTAL_CANDIDATOS_DEBITO] = 0 then
                "SOLO_DEBITO"
            else if [ID_CREDITO] = null then
                "AMBIGUO"
            else
                "CONCILIADO",
        type text
    ),

    AgregarEmisor = Table.AddColumn(
        ClasificarDebitos,
        "ID_EMISOR",
        each [ID_EMISOR_CARTOLA],
        type text
    ),

    AgregarBeneficiario = Table.AddColumn(
        AgregarEmisor,
        "ID_BENEFICIARIO",
        each [ID_CONTRAPARTE_ENDOSO],
        type text
    ),

    AgregarMontoDebito = Table.AddColumn(
        AgregarBeneficiario,
        "MONTO_OPERACION",
        each [#"MONTO MOV."],
        type number
    ),

    AgregarFechaDebitoResultado = Table.AddColumn(
        AgregarMontoDebito,
        "FECHA_DEBITO",
        each [#"FECHA DE MOV."],
        type datetime
    ),

    AgregarCodigoDebitoResultado = Table.AddColumn(
        AgregarFechaDebitoResultado,
        "CODIGO_DEBITO",
        each [#"CODIGO MOV."]
    ),

    ResultadoDebitos = Table.SelectColumns(
        AgregarCodigoDebitoResultado,
        {
            "ESTADO_CONCILIACION",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "MONTO_OPERACION",
            "FECHA_DEBITO",
            "FECHA_CREDITO",
            "DIFERENCIA_MINUTOS",
            "CODIGO_DEBITO",
            "CODIGO_CREDITO",
            "TOTAL_CANDIDATOS_DEBITO",
            "TOTAL_CANDIDATOS_CREDITO"
        }
    ),

    // =====================================================
    // 12. IDENTIFICAR CREDITOS SIN PAREJA CONFIRMADA
    // =====================================================
    UnirConteoCreditoOriginal = Table.NestedJoin(
        CreditosIndexados,
        {"ID_CREDITO"},
        ConteoCreditos,
        {"ID_CREDITO"},
        "CONTEO",
        JoinKind.LeftOuter
    ),

    ExpandirConteoCreditoOriginal = Table.ExpandTableColumn(
        UnirConteoCreditoOriginal,
        "CONTEO",
        {"TOTAL_CANDIDATOS_CREDITO"},
        {"TOTAL_CANDIDATOS_CREDITO"}
    ),

    ReemplazarConteoNuloCredito = Table.ReplaceValue(
        ExpandirConteoCreditoOriginal,
        null,
        0,
        Replacer.ReplaceValue,
        {"TOTAL_CANDIDATOS_CREDITO"}
    ),

    FiltrarCreditosPendientes = Table.SelectRows(
        ReemplazarConteoNuloCredito,
        each
            not List.Contains(
                CreditosConfirmados,
                [ID_CREDITO]
            )
    ),

    ClasificarCreditos = Table.AddColumn(
        FiltrarCreditosPendientes,
        "ESTADO_CONCILIACION",
        each
            if [TOTAL_CANDIDATOS_CREDITO] = 0 then
                "SOLO_CREDITO"
            else
                "AMBIGUO",
        type text
    ),

    CreditoAgregarEmisor = Table.AddColumn(
        ClasificarCreditos,
        "ID_EMISOR",
        each null,
        type nullable text
    ),

    CreditoAgregarBeneficiario = Table.AddColumn(
        CreditoAgregarEmisor,
        "ID_BENEFICIARIO",
        each [ID_BENEFICIARIO_CARTOLA],
        type text
    ),

    CreditoAgregarMonto = Table.AddColumn(
        CreditoAgregarBeneficiario,
        "MONTO_OPERACION",
        each [#"MONTO MOV."],
        type number
    ),

    CreditoAgregarFechaDebito = Table.AddColumn(
        CreditoAgregarMonto,
        "FECHA_DEBITO",
        each null,
        type nullable datetime
    ),

    CreditoAgregarFecha = Table.AddColumn(
        CreditoAgregarFechaDebito,
        "FECHA_CREDITO",
        each [#"FECHA DE MOV."],
        type datetime
    ),

    CreditoAgregarDiferencia = Table.AddColumn(
        CreditoAgregarFecha,
        "DIFERENCIA_MINUTOS",
        each null,
        type nullable number
    ),

    CreditoAgregarCodigoDebito = Table.AddColumn(
        CreditoAgregarDiferencia,
        "CODIGO_DEBITO",
        each null
    ),

    CreditoAgregarCodigo = Table.AddColumn(
        CreditoAgregarCodigoDebito,
        "CODIGO_CREDITO",
        each [#"CODIGO MOV."]
    ),

    CreditoAgregarConteoDebito = Table.AddColumn(
        CreditoAgregarCodigo,
        "TOTAL_CANDIDATOS_DEBITO",
        each null,
        Int64.Type
    ),

    ResultadoCreditos = Table.SelectColumns(
        CreditoAgregarConteoDebito,
        {
            "ESTADO_CONCILIACION",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "MONTO_OPERACION",
            "FECHA_DEBITO",
            "FECHA_CREDITO",
            "DIFERENCIA_MINUTOS",
            "CODIGO_DEBITO",
            "CODIGO_CREDITO",
            "TOTAL_CANDIDATOS_DEBITO",
            "TOTAL_CANDIDATOS_CREDITO"
        }
    ),

    // =====================================================
    // 13. UNIFICAR RESULTADOS
    // =====================================================
    CombinarResultados = Table.Combine(
        {
            ResultadoDebitos,
            ResultadoCreditos
        }
    ),

    OrdenarResultados = Table.Sort(
        CombinarResultados,
        {
            {"ESTADO_CONCILIACION", Order.Ascending},
            {"ID_EMISOR", Order.Ascending},
            {"ID_BENEFICIARIO", Order.Ascending},
            {"FECHA_DEBITO", Order.Ascending}
        }
    )
in
    OrdenarResultados
