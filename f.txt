let
    Origen = qryRedTransacciones,

    FiltrarRelaciones = Table.SelectRows(
        Origen,
        each
            [#"IDENTIFICACION DEL EMISOR / ENDOSANTE"] <> null
            and [#"IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"] <> null
            and Text.Trim(Text.From([#"IDENTIFICACION DEL EMISOR / ENDOSANTE"])) <> ""
            and Text.Trim(Text.From([#"IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"])) <> ""
            and [#"IDENTIFICACION DEL EMISOR / ENDOSANTE"]
                <> [#"IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"]
    ),

    SeleccionarRelaciones = Table.SelectColumns(
        FiltrarRelaciones,
        {
            "IDENTIFICACION EMPLEADO (EF)",
            "RUC SOCIAL EMPLEADOR",
            "FECHA DE MOV.",
            "CODIGO MOV.",
            "CODIGO LIQUID.",
            "IDENTIFICACION DEL EMISOR / ENDOSANTE",
            "NOMBRE DEL EMISOR / ENDOSANTE",
            "IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION",
            "NOMBRE DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION",
            "EN TRAMA EMISOR",
            "EN TRAMA BENEF.",
            "TIPO MOV.",
            "NOMBRE MOV.",
            "MONTO MOV.",
            "SALDO NCD",
            "CLASIFICACION_MOVIMIENTO"
        }
    )
in
    SeleccionarRelaciones
