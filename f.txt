let
    EmisorTrama =
        if [#"EN TRAMA EMISOR"] = null
        then ""
        else Text.Upper(Text.Trim(Text.From([#"EN TRAMA EMISOR"]))),

    BeneficiarioTrama =
        if [#"EN TRAMA BENEF."] = null
        then ""
        else Text.Upper(Text.Trim(Text.From([#"EN TRAMA BENEF."]))),

    Clasificacion = [CLASIFICACION_MOVIMIENTO]
in
    if EmisorTrama = "T" and BeneficiarioTrama = "T" then
        "TRANSFERENCIA_INTERNA"
    else if EmisorTrama = "T" and BeneficiarioTrama = "" then
        "SALIDA_TRAMA"
    else if EmisorTrama = "" and BeneficiarioTrama = "T" then
        "ENTRADA_TRAMA"
    else if EmisorTrama = "" and BeneficiarioTrama = "" then
        "TRANSFERENCIA_EXTERNA"
    else
        "REVISAR"
