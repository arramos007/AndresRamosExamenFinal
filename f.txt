
let
    Origen = qryModeloMovimientos,

    FiltrarEndosos = Table.SelectRows(
        Origen,
        each
            [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO"
            or [CLASIFICACION_MOVIMIENTO] = "ENDOSO_RECIBIDO"
    ),

    SeleccionarColumnas = Table.SelectColumns(
        FiltrarEndosos,
        {
            "REL",
            "RELAC.",
            "RUC SOCIAL EMPLEADOR",
            "RAZON SOCIAL EMPLEADOR",
            "IDENTIFICACION EMPLEADO (EF)",
            "NOMBRE EMPLEADO (EF)",
            "ID_TITULAR",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "FECHA DE MOV.",
            "CODIGO MOV.",
            "NOMBRE MOV.",
            "TIPO MOV.",
            "TIPO_FLUJO",
            "TIPO_OPERACION",
            "MONTO MOV.",
            "SALDO NCD",
            "EN TRAMA EMISOR",
            "EN TRAMA BENEF.",
            "CLAVE_MOVIMIENTO",
            "ESTADO_TRAZABILIDAD"
        }
    ),

    NormalizarTipo = Table.AddColumn(
        SeleccionarColumnas,
        "TIPO_CARTOLA",
        each
            Text.Upper(
                Text.Trim(
                    if [#"TIPO MOV."] = null
                    then ""
                    else Text.From([#"TIPO MOV."])
                )
            ),
        type text
    ),

    IdentificarTitularCartola = Table.AddColumn(
        NormalizarTipo,
        "ID_EMISOR_CARTOLA",
        each
            if [TIPO_CARTOLA] = "DEB"
            then [ID_TITULAR]
            else null,
        type text
    ),

    IdentificarBeneficiarioCartola = Table.AddColumn(
        IdentificarTitularCartola,
        "ID_BENEFICIARIO_CARTOLA",
        each
            if [TIPO_CARTOLA] = "CRE"
            then [ID_TITULAR]
            else null,
        type text
    ),

    NombreEmisorCartola = Table.AddColumn(
        IdentificarBeneficiarioCartola,
        "NOMBRE_EMISOR_CARTOLA",
        each
            if [TIPO_CARTOLA] = "DEB"
            then [#"NOMBRE EMPLEADO (EF)"]
            else null,
        type text
    ),

    NombreBeneficiarioCartola = Table.AddColumn(
        NombreEmisorCartola,
        "NOMBRE_BENEFICIARIO_CARTOLA",
        each
            if [TIPO_CARTOLA] = "CRE"
            then [#"NOMBRE EMPLEADO (EF)"]
            else null,
        type text
    ),

    ValidarTitularCartola = Table.AddColumn(
        NombreBeneficiarioCartola,
        "VALIDACION_LADO_CARTOLA",
        each
            if [TIPO_CARTOLA] = "DEB" then
                if [ID_TITULAR] = [ID_EMISOR]
                then "DEBITO_VALIDO"
                else "DEBITO_REVISAR_IDENTIFICADOR"
            else if [TIPO_CARTOLA] = "CRE" then
                if [ID_TITULAR] = [ID_BENEFICIARIO]
                then "CREDITO_VALIDO"
                else "CREDITO_REVISAR_IDENTIFICADOR"
            else
                "TIPO_MOVIMIENTO_NO_RECONOCIDO",
        type text
    ),

    IdentificarContraparte = Table.AddColumn(
        ValidarTitularCartola,
        "ID_CONTRAPARTE_ENDOSO",
        each
            if [TIPO_CARTOLA] = "DEB" then
                [ID_BENEFICIARIO]
            else if [TIPO_CARTOLA] = "CRE" then
                [ID_EMISOR]
            else
                null,
        type text
    ),

    OrdenarPorFecha = Table.Sort(
        IdentificarContraparte,
        {{"FECHA DE MOV.", Order.Ascending}}
    )
in
    OrdenarPorFecha
