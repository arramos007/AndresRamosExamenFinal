let
    Origen = qryResumenBeneficiariosPerfiles,

    CombinarTotales = Table.NestedJoin(
        Origen,
        {"ID_BENEFICIARIO"},
        qryBeneficiariosExternos,
        {"ID_BENEFICIARIO"},
        "TotalesBeneficiario",
        JoinKind.LeftOuter
    ),

    ExpandirTotales = Table.ExpandTableColumn(
        CombinarTotales,
        "TotalesBeneficiario",
        {"MONTO_TOTAL_RECIBIDO"},
        {"MONTO_TOTAL_BENEFICIARIO"}
    ),

    CalcularParticipacion = Table.AddColumn(
        ExpandirTotales,
        "PARTICIPACION_PORCENTUAL",
        each
            if [MONTO_TOTAL_BENEFICIARIO] = null
                or [MONTO_TOTAL_BENEFICIARIO] = 0
            then null
            else [MONTO_TOTAL] / [MONTO_TOTAL_BENEFICIARIO],
        Percentage.Type
    ),

    OrdenarResultados = Table.Sort(
        CalcularParticipacion,
        {
            {"ID_BENEFICIARIO", Order.Ascending},
            {"PARTICIPACION_PORCENTUAL", Order.Descending}
        }
    )
in
    OrdenarResultados
