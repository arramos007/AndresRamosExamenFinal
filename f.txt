qry16_ValidarEndososConciliados


let
    Origen = qry15_EndososConciliados,

    TotalFilas =
        Table.RowCount(Origen),

    DebitosDistintos =
        List.Count(
            List.Distinct(
                Origen[ID_REGISTRO_DEBITO]
            )
        ),

    CreditosDistintos =
        List.Count(
            List.Distinct(
                Origen[ID_REGISTRO_CREDITO]
            )
        ),

    ImportesIguales =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [MONTO_DEBITO] = [MONTO_CREDITO]
            )
        ),

    DiferenciasCorrectas =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each
                    [DIFERENCIA_SEGUNDOS] <= 3
                    and [DIFERENCIA_SEGUNDOS] >= 0
            )
        ),

    Resultado =
        #table(
            {"VALIDACION", "RESULTADO", "ESPERADO", "ESTADO"},
            {
                {
                    "TOTAL DE FILAS",
                    TotalFilas,
                    1338,
                    if TotalFilas = 1338 then "CORRECTO" else "REVISAR"
                },
                {
                    "DEBITOS DISTINTOS",
                    DebitosDistintos,
                    1338,
                    if DebitosDistintos = 1338 then "CORRECTO" else "REVISAR"
                },
                {
                    "CREDITOS DISTINTOS",
                    CreditosDistintos,
                    1338,
                    if CreditosDistintos = 1338 then "CORRECTO" else "REVISAR"
                },
                {
                    "IMPORTES IGUALES",
                    ImportesIguales,
                    1338,
                    if ImportesIguales = 1338 then "CORRECTO" else "REVISAR"
                },
                {
                    "DIFERENCIAS HASTA 3 SEGUNDOS",
                    DiferenciasCorrectas,
                    1338,
                    if DiferenciasCorrectas = 1338 then "CORRECTO" else "REVISAR"
                }
            }
        )
in
    Resultado
