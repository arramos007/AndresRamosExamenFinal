
let
    Origen =
        Table.Buffer(qry22_ComparacionEntidades),

    Indicadores = {
        "TOTAL_RECIBIDO",
        "TOTAL_REALIZADO",
        "DIFERENCIA_RECIBIDO_REALIZADO",
        "CANTIDAD_ENDOSOS_RECIBIDOS",
        "CANTIDAD_ENDOSOS_REALIZADOS",
        "CANTIDAD_RUC_EMISORES_RELACIONADOS",
        "CANTIDAD_RUC_BENEFICIARIOS_RELACIONADOS",
        "RATIO_SALIDA_ENTRADA"
    },

    CalcularEstadisticas =
        (Nombre as text) as record =>
            let
                Valores =
                    List.RemoveNulls(
                        Table.Column(Origen, Nombre)
                    ),

                Cantidad = List.Count(Valores),

                Resultado =
                    if Cantidad = 0 then
                        [
                            INDICADOR = Nombre,
                            CANTIDAD = 0,
                            MINIMO = null,
                            MEDIANA = null,
                            MAXIMO = null,
                            PROMEDIO = null
                        ]
                    else
                        [
                            INDICADOR = Nombre,
                            CANTIDAD = Cantidad,
                            MINIMO = List.Min(Valores),
                            MEDIANA = List.Median(Valores),
                            MAXIMO = List.Max(Valores),
                            PROMEDIO = List.Average(Valores)
                        ]
            in
                Resultado,

    Registros =
        List.Transform(
            Indicadores,
            each CalcularEstadisticas(_)
        ),

    Resultado =
        Table.FromRecords(Registros),

    Tipos =
        Table.TransformColumnTypes(
            Resultado,
            {
                {"INDICADOR", type text},
                {"CANTIDAD", Int64.Type},
                {"MINIMO", type number},
                {"MEDIANA", type number},
                {"MAXIMO", type number},
                {"PROMEDIO", type number}
            }
        )
in
    Tipos
