qry42_ControlCrecimientoPrimerNivel


let
    // 1. Cargar las rutas iniciales
    EstadoInicial =
        qry38_EstadoInicialRutas,

    // 2. Expandir solamente el primer nivel
    NuevasRutas =
        fnExpandirUnNivel(
            EstadoInicial,
            qry34_IndiceConexionesPorParticipante
        ),

    // 3. Convertir las rutas generadas en tabla
    TablaRutas =
        if List.IsEmpty(NuevasRutas) then
            #table(
                {
                    "ID_RUTA_ACTUAL",
                    "ESTADO_RUTA",
                    "ID_REGISTRO_DEBITO",
                    "ID_PARTICIPANTE_INICIAL",
                    "ID_PARTICIPANTE_ACTUAL",
                    "NUMERO_SALTO",
                    "ES_CICLO"
                },
                {}
            )
        else
            Table.FromRecords(NuevasRutas),

    // 4. Agrupar por estado
    AgruparEstados =
        Table.Group(
            TablaRutas,
            {"ESTADO_RUTA"},
            {
                {
                    "CANTIDAD_RUTAS",
                    each Table.RowCount(_),
                    Int64.Type
                },
                {
                    "OPERACIONES_DISTINTAS",
                    each
                        List.Count(
                            List.Distinct(
                                [ID_REGISTRO_DEBITO]
                            )
                        ),
                    Int64.Type
                },
                {
                    "PARTICIPANTES_INICIALES",
                    each
                        List.Count(
                            List.Distinct(
                                [ID_PARTICIPANTE_INICIAL]
                            )
                        ),
                    Int64.Type
                },
                {
                    "PARTICIPANTES_DESTINO",
                    each
                        List.Count(
                            List.Distinct(
                                [ID_PARTICIPANTE_ACTUAL]
                            )
                        ),
                    Int64.Type
                }
            }
        ),

    // 5. Agregar totales generales
    TotalRutas =
        Table.RowCount(TablaRutas),

    TotalOperaciones =
        if TotalRutas = 0 then
            0
        else
            List.Count(
                List.Distinct(
                    TablaRutas[ID_REGISTRO_DEBITO]
                )
            ),

    TotalParticipantes =
        if TotalRutas = 0 then
            0
        else
            List.Count(
                List.Distinct(
                    TablaRutas[ID_PARTICIPANTE_ACTUAL]
                )
            ),

    TotalCiclos =
        if TotalRutas = 0 then
            0
        else
            Table.RowCount(
                Table.SelectRows(
                    TablaRutas,
                    each [ES_CICLO] = true
                )
            ),

    Resumen =
        #table(
            {
                "ESTADO_RUTA",
                "CANTIDAD_RUTAS",
                "OPERACIONES_DISTINTAS",
                "PARTICIPANTES_INICIALES",
                "PARTICIPANTES_DESTINO"
            },
            {
                {
                    "TOTAL_GENERAL",
                    TotalRutas,
                    TotalOperaciones,
                    null,
                    TotalParticipantes
                },
                {
                    "TOTAL_CICLOS",
                    TotalCiclos,
                    null,
                    null,
                    null
                }
            }
        ),

    // 6. Unir el resumen de estados y totales
    Resultado =
        Table.Combine(
            {
                AgruparEstados,
                Resumen
            }
        ),

    Ordenar =
        Table.Sort(
            Resultado,
            {
                {"ESTADO_RUTA", Order.Ascending}
            }
        )
in
    Ordenar
