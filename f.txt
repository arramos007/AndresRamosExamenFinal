qry33_ValidacionConexionesRutas


let
    Origen = qry32_ConexionesParaRutas,

    TotalFilas =
        Table.RowCount(Origen),

    DebitosDistintos =
        List.Count(
            List.Distinct(
                Origen[ID_REGISTRO_DEBITO]
            )
        ),

    CreditosDistintos =
        List.Count(
            List.Distinct(
                Origen[ID_REGISTRO_CREDITO]
            )
        ),

    TotalImportes =
        List.Sum(Origen[MONTO_DEBITO]),

    FilasSinOrigen =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ID_PARTICIPANTE_ORIGEN] = null
            )
        ),

    FilasSinDestino =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ID_PARTICIPANTE_DESTINO] = null
            )
        ),

    Validaciones = {
        {"TOTAL_CONEXIONES", TotalFilas, 1338},
        {"DEBITOS_DISTINTOS", DebitosDistintos, 1338},
        {"CREDITOS_DISTINTOS", CreditosDistintos, 1338},
        {"IMPORTE_TOTAL", TotalImportes, 7453839.97},
        {"ORIGENES_SIN_IDENTIFICAR", FilasSinOrigen, 0},
        {"DESTINOS_SIN_IDENTIFICAR", FilasSinDestino, 0}
    },

    Tabla =
        Table.FromRows(
            Validaciones,
            {
                "VALIDACION",
                "RESULTADO",
                "ESPERADO"
            }
        ),

    AgregarDiferencia =
        Table.AddColumn(
            Tabla,
            "DIFERENCIA",
            each [RESULTADO] - [ESPERADO],
            type number
        ),

    AgregarEstado =
        Table.AddColumn(
            AgregarDiferencia,
            "ESTADO",
            each
                if Number.Abs([DIFERENCIA]) < 0.01 then
                    "CORRECTO"
                else
                    "REVISAR",
            type text
        )
in
    AgregarEstado
