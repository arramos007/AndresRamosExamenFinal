qry43_ValidacionCronologicaPrueba


let
    Origen = qry40_PruebaExpansionNivel1,

    UnirRutaPadre =
        Table.NestedJoin(
            Origen,
            {"ID_RUTA_PADRE"},
            qry38_EstadoInicialRutas,
            {"ID_RUTA_ACTUAL"},
            "PADRE",
            JoinKind.LeftOuter
        ),

    ExpandirFechaPadre =
        Table.ExpandTableColumn(
            UnirRutaPadre,
            "PADRE",
            {"FECHA_ULTIMO_MOVIMIENTO"},
            {"FECHA_CREDITO_ANTERIOR"}
        ),

    Total =
        Table.RowCount(ExpandirFechaPadre),

    PadresIdentificados =
        Table.RowCount(
            Table.SelectRows(
                ExpandirFechaPadre,
                each [FECHA_CREDITO_ANTERIOR] <> null
            )
        ),

    AbiertasCorrectas =
        Table.RowCount(
            Table.SelectRows(
                ExpandirFechaPadre,
                each
                    [ESTADO_RUTA] = "ABIERTA"
                    and [FECHA_DEBITO] > [FECHA_CREDITO_ANTERIOR]
            )
        ),

    AmbiguasCorrectas =
        Table.RowCount(
            Table.SelectRows(
                ExpandirFechaPadre,
                each
                    [ESTADO_RUTA] = "ORDEN_TEMPORAL_AMBIGUO"
                    and [FECHA_DEBITO] = [FECHA_CREDITO_ANTERIOR]
            )
        ),

    NoVerificablesCorrectas =
        Table.RowCount(
            Table.SelectRows(
                ExpandirFechaPadre,
                each
                    [ESTADO_RUTA] = "SECUENCIA_TEMPORAL_NO_VERIFICABLE"
                    and [FECHA_DEBITO] < [FECHA_CREDITO_ANTERIOR]
            )
        ),

    CiclosCorrectos =
        Table.RowCount(
            Table.SelectRows(
                ExpandirFechaPadre,
                each
                    [ESTADO_RUTA] = "CICLO_DETECTADO"
                    and [ES_CICLO] = true
            )
        ),

    Validaciones = {
        {"TOTAL_CONEXIONES", Total, Total},
        {"PADRES_IDENTIFICADOS", PadresIdentificados, Total},
        {"ABIERTAS_CRONOLOGICAMENTE_CORRECTAS", AbiertasCorrectas,
            Table.RowCount(
                Table.SelectRows(
                    ExpandirFechaPadre,
                    each [ESTADO_RUTA] = "ABIERTA"
                )
            )
        },
        {"AMBIGUAS_CRONOLOGICAMENTE_CORRECTAS", AmbiguasCorrectas,
            Table.RowCount(
                Table.SelectRows(
                    ExpandirFechaPadre,
                    each [ESTADO_RUTA] = "ORDEN_TEMPORAL_AMBIGUO"
                )
            )
        },
        {"NO_VERIFICABLES_CORRECTAS", NoVerificablesCorrectas,
            Table.RowCount(
                Table.SelectRows(
                    ExpandirFechaPadre,
                    each [ESTADO_RUTA] = "SECUENCIA_TEMPORAL_NO_VERIFICABLE"
                )
            )
        },
        {"CICLOS_CORRECTOS", CiclosCorrectos,
            Table.RowCount(
                Table.SelectRows(
                    ExpandirFechaPadre,
                    each [ESTADO_RUTA] = "CICLO_DETECTADO"
                )
            )
        }
    },

    Tabla =
        Table.FromRows(
            Validaciones,
            {"VALIDACION", "RESULTADO", "ESPERADO"}
        ),

    AgregarEstado =
        Table.AddColumn(
            Tabla,
            "ESTADO",
            each
                if [RESULTADO] = [ESPERADO] then
                    "CORRECTO"
                else
                    "REVISAR",
            type text
        )
in
    AgregarEstado
