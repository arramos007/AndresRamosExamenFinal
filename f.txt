
let
    Origen = qryModeloMovimientos,

    RelacionesValidas = Table.SelectRows(
        Origen,
        each
            [ESTADO_TRAZABILIDAD] = "TRAZABILIDAD_COMPLETA"
            and [ID_EMISOR] <> [ID_BENEFICIARIO]
    ),

    ColumnasRelaciones = Table.SelectColumns(
        RelacionesValidas,
        {
            "CLAVE_MOVIMIENTO",
            "CODIGO MOV.",
            "FECHA DE MOV.",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "ID_TITULAR",
            "ID_CONTRAPARTE",
            "MONTO MOV.",
            "TIPO MOV.",
            "TIPO_FLUJO",
            "TIPO_OPERACION",
            "EN TRAMA EMISOR",
            "EN TRAMA BENEF."
        }
    )
in
    ColumnasRelaciones
