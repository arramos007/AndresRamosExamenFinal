
let
    Origen = qry05_MovimientosEconomicos,

    UnirParticipanteTitular =
        Table.NestedJoin(
            Origen,
            {"IDENTIFICACION EMPLEADO (EF)"},
            qry02_Participantes,
            {"IDENTIFICACION_PARTICIPANTE"},
            "PARTICIPANTE_TITULAR",
            JoinKind.LeftOuter
        ),

    ExpandirParticipanteTitular =
        Table.ExpandTableColumn(
            UnirParticipanteTitular,
            "PARTICIPANTE_TITULAR",
            {"ID_PARTICIPANTE"},
            {"ID_PARTICIPANTE_TITULAR"}
        ),

    UnirParticipanteEmisor =
        Table.NestedJoin(
            ExpandirParticipanteTitular,
            {"IDENTIFICACION DEL EMISOR / ENDOSANTE"},
            qry02_Participantes,
            {"IDENTIFICACION_PARTICIPANTE"},
            "PARTICIPANTE_EMISOR",
            JoinKind.LeftOuter
        ),

    ExpandirParticipanteEmisor =
        Table.ExpandTableColumn(
            UnirParticipanteEmisor,
            "PARTICIPANTE_EMISOR",
            {"ID_PARTICIPANTE"},
            {"ID_PARTICIPANTE_EMISOR"}
        ),

    UnirParticipanteBeneficiario =
        Table.NestedJoin(
            ExpandirParticipanteEmisor,
            {
                "IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"
            },
            qry02_Participantes,
            {"IDENTIFICACION_PARTICIPANTE"},
            "PARTICIPANTE_BENEFICIARIO",
            JoinKind.LeftOuter
        ),

    ExpandirParticipanteBeneficiario =
        Table.ExpandTableColumn(
            UnirParticipanteBeneficiario,
            "PARTICIPANTE_BENEFICIARIO",
            {"ID_PARTICIPANTE"},
            {"ID_PARTICIPANTE_BENEFICIARIO"}
        )
in
    ExpandirParticipanteBeneficiario
