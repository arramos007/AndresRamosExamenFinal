let
    Origen = qryAuditoriaDuplicadosExactos,

    Agrupar = Table.Group(
        Origen,
        {
            "IDENTIFICACION EMPLEADO (EF)",
            "CODIGO MOV."
        },
        {
            {
                "EMPRESAS",
                each List.Count(
                    List.Distinct(
                        List.RemoveNulls(
                            Table.Column(_, "RUC SOCIAL EMPLEADOR")
                        )
                    )
                ),
                Int64.Type
            },
            {
                "REPETICIONES_DISTINTAS",
                each List.Count(
                    List.Distinct(
                        List.RemoveNulls(
                            Table.Column(_, "TOTAL_REPETICIONES")
                        )
                    )
                ),
                Int64.Type
            },
            {
                "EXCEDENTES_DISTINTOS",
                each List.Count(
                    List.Distinct(
                        List.RemoveNulls(
                            Table.Column(_, "REGISTROS_EXCEDENTES")
                        )
                    )
                ),
                Int64.Type
            },
            {
                "MONTOS_DISTINTOS",
                each List.Count(
                    List.Distinct(
                        List.RemoveNulls(
                            Table.Column(_, "MONTO_EXCEDENTE_CANDIDATO")
                        )
                    )
                ),
                Int64.Type
            },
            {
                "MONTO_MINIMO",
                each List.Min(
                    List.RemoveNulls(
                        Table.Column(_, "MONTO_EXCEDENTE_CANDIDATO")
                    )
                ),
                type number
            },
            {
                "MONTO_MAXIMO",
                each List.Max(
                    List.RemoveNulls(
                        Table.Column(_, "MONTO_EXCEDENTE_CANDIDATO")
                    )
                ),
                type number
            }
        }
    ),

    FiltrarCompartidos = Table.SelectRows(
        Agrupar,
        each [EMPRESAS] > 1
    ),

    Ordenar = Table.Sort(
        FiltrarCompartidos,
        {
            {"MONTOS_DISTINTOS", Order.Descending},
            {"EXCEDENTES_DISTINTOS", Order.Descending}
        }
    )
in
    Ordenar
