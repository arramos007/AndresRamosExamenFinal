qry04_RelacionesEmpresariales


let
    Origen = qry01_RegistrosOriginales,

    SeleccionarColumnas =
        Table.SelectColumns(
            Origen,
            {
                "IDENTIFICACION EMPLEADO (EF)",
                "RUC SOCIAL EMPLEADOR",
                "REL",
                "RELAC."
            }
        ),

    RelacionesUnicas =
        Table.Distinct(
            SeleccionarColumnas
        ),

    UnirParticipantes =
        Table.NestedJoin(
            RelacionesUnicas,
            {"IDENTIFICACION EMPLEADO (EF)"},
            qry02_Participantes,
            {"IDENTIFICACION_PARTICIPANTE"},
            "PARTICIPANTE",
            JoinKind.LeftOuter
        ),

    ExpandirParticipantes =
        Table.ExpandTableColumn(
            UnirParticipantes,
            "PARTICIPANTE",
            {"ID_PARTICIPANTE"},
            {"ID_PARTICIPANTE"}
        ),

    UnirEmpresas =
        Table.NestedJoin(
            ExpandirParticipantes,
            {"RUC SOCIAL EMPLEADOR"},
            qry03_Empresas,
            {"RUC_EMPRESA"},
            "EMPRESA",
            JoinKind.LeftOuter
        ),

    ExpandirEmpresas =
        Table.ExpandTableColumn(
            UnirEmpresas,
            "EMPRESA",
            {"ID_EMPRESA"},
            {"ID_EMPRESA"}
        ),

    RenombrarColumnas =
        Table.RenameColumns(
            ExpandirEmpresas,
            {
                {"REL", "TIPO_RELACION"},
                {"RELAC.", "RELACIONES_EMPLEADO"}
            }
        ),

    OrdenarRelaciones =
        Table.Sort(
            RenombrarColumnas,
            {
                {"ID_PARTICIPANTE", Order.Ascending},
                {"ID_EMPRESA", Order.Ascending},
                {"TIPO_RELACION", Order.Ascending}
            }
        ),

    AgregarIdentificador =
        Table.AddIndexColumn(
            OrdenarRelaciones,
            "ID_RELACION",
            1,
            1,
            Int64.Type
        ),

    Resultado =
        Table.ReorderColumns(
            AgregarIdentificador,
            {
                "ID_RELACION",
                "ID_PARTICIPANTE",
                "ID_EMPRESA",
                "TIPO_RELACION",
                "RELACIONES_EMPLEADO"
            }
        )
in
    Resultado
