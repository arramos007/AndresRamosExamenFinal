qry18_ResumenConexionesEmpresariales


let
    Origen = qry17_EndososConRelacionesEmpresariales,

    AgruparConexiones =
        Table.Group(
            Origen,
            {"TIPO_CONEXION_EMPRESARIAL"},
            {
                {
                    "CANTIDAD_COMBINACIONES",
                    each Table.RowCount(_),
                    Int64.Type
                },
                {
                    "OPERACIONES_DISTINTAS",
                    each
                        List.Count(
                            List.Distinct(
                                List.RemoveNulls(
                                    [ID_OPERACION]
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "EMPRESAS_EMISOR_DISTINTAS",
                    each
                        List.Count(
                            List.Distinct(
                                List.RemoveNulls(
                                    [ID_EMPRESA_EMISOR]
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "EMPRESAS_BENEFICIARIO_DISTINTAS",
                    each
                        List.Count(
                            List.Distinct(
                                List.RemoveNulls(
                                    [ID_EMPRESA_BENEFICIARIO]
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "IMPORTE_REFERENCIAL",
                    each
                        List.Sum(
                            List.Distinct(
                                Table.SelectColumns(
                                    _,
                                    {
                                        "ID_OPERACION",
                                        "MONTO_DEBITO"
                                    }
                                )[MONTO_DEBITO]
                            )
                        ),
                    type nullable number
                }
            }
        ),

    Resultado =
        Table.Sort(
            AgruparConexiones,
            {
                {"OPERACIONES_DISTINTAS", Order.Descending}
            }
        )
in
    Resultado
