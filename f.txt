let
    Origen = qryClasificacionRedInterna,

    FiltrarCandidatos = Table.SelectRows(
        Origen,
        each
            [PERFIL_RED_PRELIMINAR] =
                "RECEPTOR_Y_ACUMULADOR_COMPATIBLE"
    ),

    CalcularPorcentajeSalida = Table.AddColumn(
        FiltrarCandidatos,
        "PORCENTAJE_SALIDA_SOBRE_RECIBIDO",
        each
            if [MONTO_RECIBIDO] = null or [MONTO_RECIBIDO] = 0
            then null
            else [MONTO_SALIDAS_EXTERNAS] / [MONTO_RECIBIDO],
        Percentage.Type
    ),

    Ordenar = Table.Sort(
        CalcularPorcentajeSalida,
        {
            {"MONTO_RECIBIDO", Order.Descending}
        }
    )
in
    Ordenar
