let
    Origen = qryAristasRed,

    FiltrarCodigo = Table.SelectRows(
        Origen,
        each Text.From([#"CODIGO MOV."]) = "45727095"
    ),

    SeleccionarColumnas = Table.SelectColumns(
        FiltrarCodigo,
        {
            "IDENTIFICACION EMPLEADO (EF)",
            "FECHA DE MOV.",
            "CODIGO MOV.",
            "IDENTIFICACION DEL EMISOR / ENDOSANTE",
            "IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION",
            "TIPO MOV.",
            "NOMBRE MOV.",
            "MONTO MOV.",
            "CLASIFICACION_MOVIMIENTO"
        }
    ),

    Ordenar = Table.Sort(
        SeleccionarColumnas,
        {{"IDENTIFICACION EMPLEADO (EF)", Order.Ascending}}
    )
in
    Ordenar
