
let
    Origen = qry28_RutasEconomicasDirectas,

    TotalFilas = Table.RowCount(Origen),

    DebitosDistintos =
        List.Count(
            List.Distinct(Origen[ID_REGISTRO_DEBITO])
        ),

    CreditosDistintos =
        List.Count(
            List.Distinct(Origen[ID_REGISTRO_CREDITO])
        ),

    ImportesIguales =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [MONTO_DEBITO] = [MONTO_CREDITO]
            )
        ),

    OrigenIdentificado =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ID_PARTICIPANTE_ORIGEN] <> null
            )
        ),

    DestinoIdentificado =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ID_PARTICIPANTE_DESTINO] <> null
            )
        ),

    Validaciones = {
        {"TOTAL_OPERACIONES", TotalFilas, 1338},
        {"DEBITOS_DISTINTOS", DebitosDistintos, 1338},
        {"CREDITOS_DISTINTOS", CreditosDistintos, 1338},
        {"IMPORTES_IGUALES", ImportesIguales, 1338},
        {"ORIGEN_IDENTIFICADO", OrigenIdentificado, 1338},
        {"DESTINO_IDENTIFICADO", DestinoIdentificado, 1338}
    },

    Tabla =
        Table.FromRows(
            Validaciones,
            {
                "VALIDACION",
                "RESULTADO",
                "ESPERADO"
            }
        ),

    AgregarEstado =
        Table.AddColumn(
            Tabla,
            "ESTADO",
            each
                if [RESULTADO] = [ESPERADO] then
                    "CORRECTO"
                else
                    "REVISAR",
            type text
        )
in
    AgregarEstado
