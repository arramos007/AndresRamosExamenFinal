let
    Origen = qryModeloMovimientos,

    FiltrarEndosos = Table.SelectRows(
        Origen,
        each
            [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO"
            or [CLASIFICACION_MOVIMIENTO] = "ENDOSO_RECIBIDO"
    ),

    SeleccionarColumnas = Table.SelectColumns(
        FiltrarEndosos,
        {
            "RUC SOCIAL EMPLEADOR",
            "RAZON SOCIAL EMPLEADOR",
            "IDENTIFICACION EMPLEADO (EF)",
            "NOMBRE EMPLEADO (EF)",
            "ID_TITULAR",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "FECHA DE MOV.",
            "CODIGO MOV.",
            "NOMBRE MOV.",
            "TIPO MOV.",
            "TIPO_FLUJO",
            "TIPO_OPERACION",
            "MONTO MOV.",
            "SALDO NCD",
            "EN TRAMA EMISOR",
            "EN TRAMA BENEF.",
            "CLAVE_MOVIMIENTO",
            "ESTADO_TRAZABILIDAD"
        }
    ),

    OrdenarPorFecha = Table.Sort(
        SeleccionarColumnas,
        {{"FECHA DE MOV.", Order.Ascending}}
    )
in
    OrdenarPorFecha
