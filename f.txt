let
    Origen = qryMovimientosClasificados,

    AgruparMovimientos = Table.Group(
        Origen,
        {
            "IDENTIFICACION EMPLEADO (EF)",
            "CODIGO MOV."
        },
        {
            {
                "TOTAL_REGISTROS_ORIGEN",
                each Table.RowCount(_),
                Int64.Type
            },
            {
                "EMPRESAS_RELACIONADAS",
                each List.Count(
                    List.Distinct(
                        List.Select(
                            List.Transform(
                                Table.Column(_, "RUC SOCIAL EMPLEADOR"),
                                each if _ = null then "" else Text.Trim(Text.From(_))
                            ),
                            each _ <> ""
                        )
                    )
                ),
                Int64.Type
            },
            {
                "MOVIMIENTO",
                each Table.First(_),
                type record
            }
        }
    ),

    ColumnasMovimiento = List.RemoveItems(
        Table.ColumnNames(Origen),
        {
            "IDENTIFICACION EMPLEADO (EF)",
            "CODIGO MOV."
        }
    ),

    ExpandirMovimiento = Table.ExpandRecordColumn(
        AgruparMovimientos,
        "MOVIMIENTO",
        ColumnasMovimiento,
        ColumnasMovimiento
    )
in
    ExpandirMovimiento
