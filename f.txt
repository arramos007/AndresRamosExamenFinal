qry07_MovimientosPorEmpresa


let
    Origen = qry06_MovimientosConParticipantes,

    UnirRelacionesEmpresariales =
        Table.NestedJoin(
            Origen,
            {"ID_PARTICIPANTE_TITULAR"},
            qry04_RelacionesEmpresariales,
            {"ID_PARTICIPANTE"},
            "RELACIONES_EMPRESARIALES",
            JoinKind.LeftOuter
        ),

    ExpandirRelacionesEmpresariales =
        Table.ExpandTableColumn(
            UnirRelacionesEmpresariales,
            "RELACIONES_EMPRESARIALES",
            {
                "ID_RELACION",
                "ID_EMPRESA",
                "TIPO_RELACION",
                "RELACIONES_EMPLEADO"
            },
            {
                "ID_RELACION",
                "ID_EMPRESA",
                "TIPO_RELACION",
                "RELACIONES_EMPLEADO"
            }
        ),

    Resultado =
        Table.ReorderColumns(
            ExpandirRelacionesEmpresariales,
            {
                "ID_REGISTRO_ORIGEN",
                "ID_RELACION",
                "ID_PARTICIPANTE_TITULAR",
                "ID_EMPRESA",
                "TIPO_RELACION",
                "RELACIONES_EMPLEADO"
            }
            &
            List.RemoveItems(
                Table.ColumnNames(ExpandirRelacionesEmpresariales),
                {
                    "ID_REGISTRO_ORIGEN",
                    "ID_RELACION",
                    "ID_PARTICIPANTE_TITULAR",
                    "ID_EMPRESA",
                    "TIPO_RELACION",
                    "RELACIONES_EMPLEADO"
                }
            )
        )
in
    Resultado
