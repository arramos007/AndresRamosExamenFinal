qry35_ValidacionIndiceConexiones


let
    Origen = qry34_IndiceConexionesPorParticipante,

    CantidadParticipantes =
        Table.RowCount(Origen),

    TotalSalidas =
        List.Sum(Origen[CANTIDAD_SALIDAS]),

    ParticipantesConSalidasMultiples =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [CANTIDAD_SALIDAS] > 1
            )
        ),

    TotalTablasAnidadas =
        List.Sum(
            List.Transform(
                Origen[CONEXIONES_SALIDA],
                each Table.RowCount(_)
            )
        ),

    Validaciones = {
        {
            "PARTICIPANTES_CON_SALIDAS",
            CantidadParticipantes,
            CantidadParticipantes
        },
        {
            "TOTAL_SALIDAS",
            TotalSalidas,
            1338
        },
        {
            "TOTAL_REGISTROS_TABLAS_ANIDADAS",
            TotalTablasAnidadas,
            1338
        },
        {
            "PARTICIPANTES_CON_SALIDAS_MULTIPLES",
            ParticipantesConSalidasMultiples,
            ParticipantesConSalidasMultiples
        }
    },

    Tabla =
        Table.FromRows(
            Validaciones,
            {
                "VALIDACION",
                "RESULTADO",
                "ESPERADO"
            }
        ),

    AgregarEstado =
        Table.AddColumn(
            Tabla,
            "ESTADO",
            each
                if [RESULTADO] = [ESPERADO] then
                    "CORRECTO"
                else
                    "REVISAR",
            type text
        )
in
    AgregarEstado
