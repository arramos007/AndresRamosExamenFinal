
let
    Origen = qryTrazabilidadN1,

    BeneficiariosFinales = Table.SelectRows(
        Origen,
        each
            let
                Normalizar = (Valor as any) as text =>
                    if Valor = null then ""
                    else Text.Trim(Text.From(Valor)),

                Clasificacion = Text.Upper(
                    Normalizar([CLASIFICACION_MOVIMIENTO])
                ),

                Tipo = Text.Upper(
                    Normalizar([#"TIPO MOV."])
                ),

                TramaEmisor = Text.Upper(
                    Normalizar([#"EN TRAMA EMISOR"])
                ),

                TramaBeneficiario = Text.Upper(
                    Normalizar([#"EN TRAMA BENEF."])
                ),

                RucPropietario = Normalizar(
                    [#"IDENTIFICACION EMPLEADO (EF)"]
                ),

                RucEmisor = Normalizar(
                    [#"IDENTIFICACION DEL EMISOR / ENDOSANTE"]
                ),

                RucBeneficiario = Normalizar(
                    [#"IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"]
                )
            in
                Clasificacion = "ENDOSO_REALIZADO"
                and Tipo = "DEB"
                and TramaEmisor = "T"
                and TramaBeneficiario = ""
                and RucPropietario <> ""
                and RucEmisor = RucPropietario
                and RucBeneficiario <> ""
    )
in
    BeneficiariosFinales
