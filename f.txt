qry44_ControlCrecimientoSegundoNivel


let
    // 1. Obtener las rutas iniciales
    EstadoInicial =
        qry38_EstadoInicialRutas,

    // 2. Expandir el primer nivel
    PrimeraExpansionLista =
        fnExpandirUnNivel(
            EstadoInicial,
            qry34_IndiceConexionesPorParticipante
        ),

    PrimeraExpansion =
        if List.IsEmpty(PrimeraExpansionLista) then
            #table({}, {})
        else
            Table.FromRecords(PrimeraExpansionLista),

    // 3. Conservar solamente rutas abiertas
    RutasAbiertas =
        Table.SelectRows(
            PrimeraExpansion,
            each [ESTADO_RUTA] = "ABIERTA"
        ),

    // 4. Expandir el segundo nivel
    SegundaExpansionLista =
        fnExpandirUnNivel(
            RutasAbiertas,
            qry34_IndiceConexionesPorParticipante
        ),

    SegundaExpansion =
        if List.IsEmpty(SegundaExpansionLista) then
            #table(
                {
                    "ESTADO_RUTA",
                    "ID_REGISTRO_DEBITO",
                    "ID_PARTICIPANTE_INICIAL",
                    "ID_PARTICIPANTE_ACTUAL",
                    "NUMERO_SALTO",
                    "ES_CICLO"
                },
                {}
            )
        else
            Table.FromRecords(SegundaExpansionLista),

    // 5. Resumir por estado
    AgruparEstados =
        Table.Group(
            SegundaExpansion,
            {"ESTADO_RUTA"},
            {
                {
                    "CANTIDAD_RUTAS",
                    each Table.RowCount(_),
                    Int64.Type
                },
                {
                    "OPERACIONES_DISTINTAS",
                    each
                        List.Count(
                            List.Distinct(
                                [ID_REGISTRO_DEBITO]
                            )
                        ),
                    Int64.Type
                },
                {
                    "PARTICIPANTES_INICIALES",
                    each
                        List.Count(
                            List.Distinct(
                                [ID_PARTICIPANTE_INICIAL]
                            )
                        ),
                    Int64.Type
                },
                {
                    "PARTICIPANTES_DESTINO",
                    each
                        List.Count(
                            List.Distinct(
                                [ID_PARTICIPANTE_ACTUAL]
                            )
                        ),
                    Int64.Type
                }
            }
        ),

    // 6. Totales del segundo nivel
    TotalRutas =
        Table.RowCount(SegundaExpansion),

    TotalOperaciones =
        if TotalRutas = 0 then
            0
        else
            List.Count(
                List.Distinct(
                    SegundaExpansion[ID_REGISTRO_DEBITO]
                )
            ),

    TotalCiclos =
        if TotalRutas = 0 then
            0
        else
            Table.RowCount(
                Table.SelectRows(
                    SegundaExpansion,
                    each [ES_CICLO] = true
                )
            ),

    Resumen =
        #table(
            {
                "ESTADO_RUTA",
                "CANTIDAD_RUTAS",
                "OPERACIONES_DISTINTAS",
                "PARTICIPANTES_INICIALES",
                "PARTICIPANTES_DESTINO"
            },
            {
                {
                    "TOTAL_GENERAL",
                    TotalRutas,
                    TotalOperaciones,
                    null,
                    null
                },
                {
                    "TOTAL_CICLOS",
                    TotalCiclos,
                    null,
                    null,
                    null
                }
            }
        ),

    Resultado =
        Table.Combine(
            {
                AgruparEstados,
                Resumen
            }
        ),

    Ordenar =
        Table.Sort(
            Resultado,
            {
                {"ESTADO_RUTA", Order.Ascending}
            }
        )
in
    Ordenar
