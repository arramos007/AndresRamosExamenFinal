qry03_Empresas


let
    Origen = qry01_RegistrosOriginales,

    TiposTexto =
        Table.TransformColumnTypes(
            Origen,
            {
                {"RUC SOCIAL EMPLEADOR", type text},
                {"RAZON SOCIAL EMPLEADOR", type text}
            }
        ),

    AgruparEmpresas =
        Table.Group(
            TiposTexto,
            {"RUC SOCIAL EMPLEADOR"},
            {
                {
                    "RAZON_SOCIAL",
                    each List.First(
                        List.RemoveNulls(
                            [RAZON SOCIAL EMPLEADOR]
                        ),
                        null
                    ),
                    type nullable text
                }
            }
        ),

    RenombrarRUC =
        Table.RenameColumns(
            AgruparEmpresas,
            {
                {"RUC SOCIAL EMPLEADOR", "RUC_EMPRESA"}
            }
        ),

    OrdenarEmpresas =
        Table.Sort(
            RenombrarRUC,
            {
                {"RUC_EMPRESA", Order.Ascending}
            }
        ),

    AgregarIdentificador =
        Table.AddIndexColumn(
            OrdenarEmpresas,
            "ID_EMPRESA",
            1,
            1,
            Int64.Type
        ),

    Resultado =
        Table.ReorderColumns(
            AgregarIdentificador,
            {
                "ID_EMPRESA",
                "RUC_EMPRESA",
                "RAZON_SOCIAL"
            }
        )
in
    Resultado
