let
    Origen = qryAuditoriaDuplicadosExactos,

    Agrupar = Table.Group(
        Origen,
        {"CODIGO MOV."},
        {
            {
                "TOTAL_REGISTROS",
                each Table.RowCount(_),
                Int64.Type
            },
            {
                "REGISTROS_DISTINTOS",
                each Table.RowCount(Table.Distinct(_)),
                Int64.Type
            },
            {
                "PROPIETARIOS_DISTINTOS",
                each List.Count(
                    List.Distinct(
                        Table.Column(_, "IDENTIFICACION EMPLEADO (EF)")
                    )
                ),
                Int64.Type
            },
            {
                "EMPRESAS_DISTINTAS",
                each List.Count(
                    List.Distinct(
                        Table.Column(_, "RUC SOCIAL EMPLEADOR")
                    )
                ),
                Int64.Type
            },
            {
                "ACTOS_ADMINISTRATIVOS",
                each List.Count(
                    List.Distinct(
                        List.RemoveNulls(
                            Table.Column(_, "ACTO ADMINIS.")
                        )
                    )
                ),
                Int64.Type
            },
            {
                "FECHAS_DISTINTAS",
                each List.Count(
                    List.Distinct(
                        Table.Column(_, "FECHA DE MOV.")
                    )
                ),
                Int64.Type
            },
            {
                "REPETICIONES_EXACTAS",
                each Table.RowCount(_) -
                     Table.RowCount(Table.Distinct(_)),
                Int64.Type
            }
        }
    ),

    Ordenar = Table.Sort(
        Agrupar,
        {
            {"REPETICIONES_EXACTAS", Order.Descending},
            {"TOTAL_REGISTROS", Order.Descending}
        }
    )
in
    Ordenar
