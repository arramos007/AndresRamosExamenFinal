let
    Original = qryDetalleUtilizaciones,

    Conciliado = qryUtilizacionesConciliadas,

    TotalOriginal = List.Sum(
        List.RemoveNulls(
            Table.Column(Original, "MONTO MOV.")
        )
    ),

    TotalConciliado = List.Sum(
        List.RemoveNulls(
            Table.Column(Conciliado, "MONTO_OPERACION_CONCILIADO")
        )
    ),

    RegistrosOriginales = Table.RowCount(Original),

    OperacionesConciliadas = Table.RowCount(Conciliado),

    Resultado = #table(
        {
            "TOTAL_REGISTROS_ORIGINALES",
            "TOTAL_OPERACIONES_CONCILIADAS",
            "MONTO_TOTAL_ORIGINAL",
            "MONTO_TOTAL_CONCILIADO",
            "DIFERENCIA_MONETARIA",
            "REGISTROS_RECONCILIADOS"
        },
        {
            {
                RegistrosOriginales,
                OperacionesConciliadas,
                TotalOriginal,
                TotalConciliado,
                TotalOriginal - TotalConciliado,
                RegistrosOriginales - OperacionesConciliadas
            }
        }
    )
in
    Resultado
