let
    Origen = qryRankingBeneficiariosAcumuladores,

    CombinarParticipantes = Table.NestedJoin(
        Origen,
        {"ID_BENEFICIARIO"},
        qryResumenParticipantes,
        {"ID_TITULAR"},
        "DatosParticipante",
        JoinKind.LeftOuter
    ),

    ExpandirParticipantes = Table.ExpandTableColumn(
        CombinarParticipantes,
        "DatosParticipante",
        {
            "TOTAL_ENTRADAS_INTERNAS",
            "TOTAL_SALIDAS_EXTERNAS",
            "MONTO_SALIDAS_EXTERNAS"
        },
        {
            "ENTRADAS_INTERNAS_COMO_TITULAR",
            "SALIDAS_EXTERNAS_COMO_TITULAR",
            "MONTO_SALIDAS_COMO_TITULAR"
        }
    ),

    ClasificarReceptor = Table.AddColumn(
        ExpandirParticipantes,
        "COMPORTAMIENTO_RECEPTOR",
        each
            if [SALIDAS_EXTERNAS_COMO_TITULAR] = null then
                "NO OBSERVADO COMO TITULAR"
            else if [SALIDAS_EXTERNAS_COMO_TITULAR] > 0 then
                "RECEPTOR CON SALIDAS EXTERNAS"
            else
                "RECEPTOR SIN SALIDAS EXTERNAS OBSERVADAS",
        type text
    )
in
    ClasificarReceptor
