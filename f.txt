let
    Origen = qryModeloMovimientos,

    FilasFiltradas = Table.SelectRows(
        Origen,
        each
            [TIPO_OPERACION] = "SALIDA_A_EXTERNO"
            and [ID_BENEFICIARIO] = "1790368718001"
    ),

    MovimientosUnicos = Table.Distinct(
        FilasFiltradas,
        {"ID_TITULAR", "CLAVE_MOVIMIENTO"}
    ),

    AgruparParticipantes = Table.Group(
        MovimientosUnicos,
        {"ID_TITULAR"},
        {
            {
                "TOTAL_MOVIMIENTOS",
                each Table.RowCount(_),
                Int64.Type
            },
            {
                "MONTO_TOTAL",
                each List.Sum(List.RemoveNulls([#"MONTO MOV."])),
                type nullable number
            },
            {
                "FECHA_PRIMERA_SALIDA",
                each List.Min(List.RemoveNulls([#"FECHA DE MOV."])),
                type nullable datetime
            },
            {
                "FECHA_ULTIMA_SALIDA",
                each List.Max(List.RemoveNulls([#"FECHA DE MOV."])),
                type nullable datetime
            }
        }
    ),

    OrdenarResultados = Table.Sort(
        AgruparParticipantes,
        {{"MONTO_TOTAL", Order.Descending}}
    )
in
    OrdenarResultados
