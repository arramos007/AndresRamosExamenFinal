let
    Origen = qryTrazabilidadN1,

    BeneficiariosFinales =
        Table.SelectRows(
            Origen,
            each
                [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO"
                and Text.Trim(
                    Text.From(
                        if [#"EN TRAMA EMISOR"] = null
                        then ""
                        else [#"EN TRAMA EMISOR"]
                    )
                ) = "T"
                and Text.Trim(
                    Text.From(
                        if [#"EN TRAMA BENEF."] = null
                        then ""
                        else [#"EN TRAMA BENEF."]
                    )
                ) = ""
                and [#"IDENTIFICACION DEL EMISOR / ENDOSANTE"]
                    <> [#"IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"]
        )
in
    BeneficiariosFinales
