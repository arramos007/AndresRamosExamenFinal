qry01_RegistrosOriginales


let
    Origen = qryBase,

    ColumnasOriginales = Table.ColumnNames(Origen),

    ValidarCantidadCampos =
        if List.Count(ColumnasOriginales) <> 23 then
            error "La fuente debe contener exactamente 23 campos."
        else
            Origen,

    AgregarIdentificador =
        Table.AddIndexColumn(
            ValidarCantidadCampos,
            "ID_REGISTRO_ORIGEN",
            1,
            1,
            Int64.Type
        )
in
    AgregarIdentificador
