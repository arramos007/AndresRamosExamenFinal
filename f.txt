qry46_ControlCrecimientoCuartoNivel


let
    Indice =
        qry34_IndiceConexionesPorParticipante,

    EstadoInicial =
        qry38_EstadoInicialRutas,

    // NIVEL 1
    Nivel1Lista =
        fnExpandirUnNivel(EstadoInicial, Indice),

    Nivel1 =
        if List.IsEmpty(Nivel1Lista) then
            #table({}, {})
        else
            Table.FromRecords(Nivel1Lista),

    AbiertasNivel1 =
        Table.SelectRows(
            Nivel1,
            each [ESTADO_RUTA] = "ABIERTA"
        ),

    // NIVEL 2
    Nivel2Lista =
        fnExpandirUnNivel(AbiertasNivel1, Indice),

    Nivel2 =
        if List.IsEmpty(Nivel2Lista) then
            #table({}, {})
        else
            Table.FromRecords(Nivel2Lista),

    AbiertasNivel2 =
        Table.SelectRows(
            Nivel2,
            each [ESTADO_RUTA] = "ABIERTA"
        ),

    // NIVEL 3
    Nivel3Lista =
        fnExpandirUnNivel(AbiertasNivel2, Indice),

    Nivel3 =
        if List.IsEmpty(Nivel3Lista) then
            #table({}, {})
        else
            Table.FromRecords(Nivel3Lista),

    AbiertasNivel3 =
        Table.SelectRows(
            Nivel3,
            each [ESTADO_RUTA] = "ABIERTA"
        ),

    // NIVEL 4
    Nivel4Lista =
        fnExpandirUnNivel(AbiertasNivel3, Indice),

    Nivel4 =
        if List.IsEmpty(Nivel4Lista) then
            #table(
                {
                    "ESTADO_RUTA",
                    "ID_REGISTRO_DEBITO",
                    "ID_PARTICIPANTE_INICIAL",
                    "ID_PARTICIPANTE_ACTUAL",
                    "NUMERO_SALTO",
                    "ES_CICLO"
                },
                {}
            )
        else
            Table.FromRecords(Nivel4Lista),

    // AGRUPAR RESULTADOS
    Agrupar =
        Table.Group(
            Nivel4,
            {"ESTADO_RUTA"},
            {
                {
                    "CANTIDAD_RUTAS",
                    each Table.RowCount(_),
                    Int64.Type
                },
                {
                    "OPERACIONES_DISTINTAS",
                    each List.Count(
                        List.Distinct([ID_REGISTRO_DEBITO])
                    ),
                    Int64.Type
                },
                {
                    "PARTICIPANTES_INICIALES",
                    each List.Count(
                        List.Distinct([ID_PARTICIPANTE_INICIAL])
                    ),
                    Int64.Type
                },
                {
                    "PARTICIPANTES_DESTINO",
                    each List.Count(
                        List.Distinct([ID_PARTICIPANTE_ACTUAL])
                    ),
                    Int64.Type
                }
            }
        ),

    TotalRutas =
        Table.RowCount(Nivel4),

    TotalOperaciones =
        if TotalRutas = 0 then
            0
        else
            List.Count(
                List.Distinct(
                    Nivel4[ID_REGISTRO_DEBITO]
                )
            ),

    TotalCiclos =
        if TotalRutas = 0 then
            0
        else
            Table.RowCount(
                Table.SelectRows(
                    Nivel4,
                    each [ES_CICLO] = true
                )
            ),

    Resumen =
        #table(
            {
                "ESTADO_RUTA",
                "CANTIDAD_RUTAS",
                "OPERACIONES_DISTINTAS",
                "PARTICIPANTES_INICIALES",
                "PARTICIPANTES_DESTINO"
            },
            {
                {
                    "TOTAL_GENERAL",
                    TotalRutas,
                    TotalOperaciones,
                    null,
                    null
                },
                {
                    "TOTAL_CICLOS",
                    TotalCiclos,
                    null,
                    null,
                    null
                }
            }
        ),

    Resultado =
        Table.Combine({Agrupar, Resumen})
in
    Resultado
