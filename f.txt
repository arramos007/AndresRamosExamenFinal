qry32_ConexionesParaRutas


let
    Origen =
        qry28_RutasEconomicasDirectas,

    SeleccionarColumnas =
        Table.SelectColumns(
            Origen,
            {
                "ID_RUTA",
                "ID_REGISTRO_DEBITO",
                "ID_REGISTRO_CREDITO",
                "ID_PARTICIPANTE_ORIGEN",
                "ID_PARTICIPANTE_DESTINO",
                "FECHA_DEBITO",
                "FECHA_CREDITO",
                "MONTO_DEBITO",
                "MONTO_CREDITO",
                "DIFERENCIA_SEGUNDOS",
                "NIVEL_EVIDENCIA",
                "ES_CICLO"
            }
        ),

    Tipos =
        Table.TransformColumnTypes(
            SeleccionarColumnas,
            {
                {"ID_RUTA", Int64.Type},
                {"ID_REGISTRO_DEBITO", Int64.Type},
                {"ID_REGISTRO_CREDITO", Int64.Type},
                {"ID_PARTICIPANTE_ORIGEN", Int64.Type},
                {"ID_PARTICIPANTE_DESTINO", Int64.Type},
                {"FECHA_DEBITO", type datetime},
                {"FECHA_CREDITO", type datetime},
                {"MONTO_DEBITO", type number},
                {"MONTO_CREDITO", type number},
                {"DIFERENCIA_SEGUNDOS", Int64.Type}
            }
        ),

    OrdenarConexiones =
        Table.Sort(
            Tipos,
            {
                {"ID_PARTICIPANTE_ORIGEN", Order.Ascending},
                {"FECHA_DEBITO", Order.Ascending},
                {"ID_REGISTRO_DEBITO", Order.Ascending}
            }
        ),

    Resultado =
        Table.Buffer(OrdenarConexiones)
in
    Resultado
