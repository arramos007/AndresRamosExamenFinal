qry14_ResumenConciliacion


let
    Origen = qry13_ConciliacionEndosos,

    AgruparEstados =
        Table.Group(
            Origen,
            {"ESTADO_CONCILIACION"},
            {
                {
                    "CANTIDAD_FILAS",
                    each Table.RowCount(_),
                    Int64.Type
                },
                {
                    "DEBITOS_DISTINTOS",
                    each
                        List.Count(
                            List.Distinct(
                                List.RemoveNulls(
                                    [ID_REGISTRO_DEBITO]
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "CREDITOS_DISTINTOS",
                    each
                        List.Count(
                            List.Distinct(
                                List.RemoveNulls(
                                    [ID_REGISTRO_CREDITO]
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "IMPORTE_DEBITOS",
                    each
                        List.Sum(
                            List.Distinct(
                                List.RemoveNulls(
                                    [MONTO_DEBITO]
                                )
                            )
                        ),
                    type nullable number
                },
                {
                    "IMPORTE_CREDITOS",
                    each
                        List.Sum(
                            List.Distinct(
                                List.RemoveNulls(
                                    [MONTO_CREDITO]
                                )
                            )
                        ),
                    type nullable number
                }
            }
        ),

    Resultado =
        Table.Sort(
            AgruparEstados,
            {
                {"ESTADO_CONCILIACION", Order.Ascending}
            }
        )
in
    Resultado
