let
    Origen = qryModeloMovimientos,

    FiltrarEntradas = Table.SelectRows(
        Origen,
        each
            [ID_TITULAR] = "0915842710001"
            and [TIPO_OPERACION] = "RECEPCION_INTERNA"
            and [TIPO_FLUJO] = "INTERNO"
            and [ESTADO_TRAZABILIDAD] = "TRAZABILIDAD_COMPLETA"
    ),

    MovimientosUnicos = Table.Distinct(
        FiltrarEntradas,
        {"CLAVE_MOVIMIENTO"}
    ),

    SeleccionarColumnas = Table.SelectColumns(
        MovimientosUnicos,
        {
            "CLAVE_MOVIMIENTO",
            "FECHA DE MOV.",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "ID_TITULAR",
            "NOMBRE DEL EMISOR / ENDOSANTE",
            "MONTO MOV.",
            "SALDO NCD"
        }
    ),

    OrdenarPorFecha = Table.Sort(
        SeleccionarColumnas,
        {{"FECHA DE MOV.", Order.Ascending}}
    )
in
    OrdenarPorFecha
