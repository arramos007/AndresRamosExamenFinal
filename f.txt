qry49_ValidacionRutasCompletas


let
    Origen = qry48_RutasCompletas,

    AgruparNiveles =
        Table.Group(
            Origen,
            {"NIVEL_EXPANSION"},
            {
                {
                    "CANTIDAD_RUTAS",
                    each Table.RowCount(_),
                    Int64.Type
                },
                {
                    "OPERACIONES_DISTINTAS",
                    each List.Count(
                        List.Distinct([ID_REGISTRO_DEBITO])
                    ),
                    Int64.Type
                },
                {
                    "PARTICIPANTES_INICIALES",
                    each List.Count(
                        List.Distinct([ID_PARTICIPANTE_INICIAL])
                    ),
                    Int64.Type
                }
            }
        ),

    Ordenar =
        Table.Sort(
            AgruparNiveles,
            {{"NIVEL_EXPANSION", Order.Ascending}}
        ),

    Total =
        #table(
            {"NIVEL_EXPANSION", "CANTIDAD_RUTAS", "OPERACIONES_DISTINTAS", "PARTICIPANTES_INICIALES"},
            {
                {
                    "TOTAL",
                    Table.RowCount(Origen),
                    List.Count(List.Distinct(Origen[ID_REGISTRO_DEBITO])),
                    List.Count(List.Distinct(Origen[ID_PARTICIPANTE_INICIAL]))
                }
            }
        ),

    Resultado =
        Table.Combine({Ordenar, Total})
in
    Resultado
