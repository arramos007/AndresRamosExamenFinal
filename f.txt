
let
    Origen = qryPerfilParticipantes,

    MovimientosRelevantes = Table.SelectRows(
        Origen,
        each
            [TIPO_MOVIMIENTO_RED] = "ENTRADA_DESDE_PARTICIPANTE"
            or [TIPO_MOVIMIENTO_RED] = "SALIDA_A_EXTERNO"
            or [TIPO_MOVIMIENTO_RED] = "SALIDA_A_PARTICIPANTE"
    ),

    ColumnasSecuencia = Table.SelectColumns(
        MovimientosRelevantes,
        {
            "ID_TITULAR",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "ID_CONTRAPARTE",
            "FECHA DE MOV.",
            "CODIGO MOV.",
            "MONTO MOV.",
            "TIPO_MOVIMIENTO_RED",
            "CLAVE_MOVIMIENTO",
            "ESTADO_TRAZABILIDAD"
        }
    ),

    OrdenarSecuencia = Table.Sort(
        ColumnasSecuencia,
        {
            {"ID_TITULAR", Order.Ascending},
            {"FECHA DE MOV.", Order.Ascending}
        }
    )
in
    OrdenarSecuencia
