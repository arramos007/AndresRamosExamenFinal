
let
    Origen = qryDetalleEndosos,

    // =====================================================
    // 1. PARAMETRO DE CONCILIACION
    // =====================================================
    ToleranciaMinutos = 10,

    // =====================================================
    // 2. SEPARAR DEBITOS Y CREDITOS
    // =====================================================
    DebitosOrigen = Table.SelectRows(
        Origen,
        each
            [TIPO_CARTOLA] = "DEB"
            and [ID_EMISOR_CARTOLA] <> null
            and [ID_CONTRAPARTE_ENDOSO] <> null
            and [#"MONTO MOV."] <> null
            and [#"FECHA DE MOV."] <> null
    ),

    CreditosOrigen = Table.SelectRows(
        Origen,
        each
            [TIPO_CARTOLA] = "CRE"
            and [ID_BENEFICIARIO_CARTOLA] <> null
            and [#"MONTO MOV."] <> null
            and [#"FECHA DE MOV."] <> null
    ),

    // =====================================================
    // 3. ELIMINAR DUPLICADOS ADMINISTRATIVOS
    //    REL y empresa no forman parte de la operacion
    // =====================================================
    DebitosUnicos = Table.Distinct(
        DebitosOrigen,
        {
            "ID_EMISOR_CARTOLA",
            "ID_CONTRAPARTE_ENDOSO",
            "MONTO MOV.",
            "FECHA DE MOV.",
            "CODIGO MOV."
        }
    ),

    CreditosUnicos = Table.Distinct(
        CreditosOrigen,
        {
            "ID_BENEFICIARIO_CARTOLA",
            "MONTO MOV.",
            "FECHA DE MOV.",
            "CODIGO MOV."
        }
    ),

    // =====================================================
    // 4. IDENTIFICADORES INTERNOS
    // =====================================================
    DebitosIndexados = Table.AddIndexColumn(
        DebitosUnicos,
        "ID_DEBITO",
        1,
        1,
        Int64.Type
    ),

    CreditosIndexados = Table.AddIndexColumn(
        CreditosUnicos,
        "ID_CREDITO",
        1,
        1,
        Int64.Type
    ),

    // =====================================================
    // 5. BUSCAR CREDITOS CANDIDATOS PARA CADA DEBITO
    // =====================================================
    BuscarCandidatos = Table.AddColumn(
        DebitosIndexados,
        "CANDIDATOS",
        each
            let
                Debito = _,

                Candidatos = Table.SelectRows(
                    CreditosIndexados,
                    (Credito) =>
                        Credito[ID_BENEFICIARIO_CARTOLA]
                            = Debito[ID_CONTRAPARTE_ENDOSO]

                        and Credito[#"MONTO MOV."]
                            = Debito[#"MONTO MOV."]

                        and Number.Abs(
                            Duration.TotalMinutes(
                                Credito[#"FECHA DE MOV."]
                                - Debito[#"FECHA DE MOV."]
                            )
                        ) <= ToleranciaMinutos
                )
            in
                Candidatos
    ),

    ContarCandidatosDebito = Table.AddColumn(
        BuscarCandidatos,
        "TOTAL_CANDIDATOS_DEBITO",
        each Table.RowCount([CANDIDATOS]),
        Int64.Type
    ),

    // =====================================================
    // 6. CONTAR CUANTOS DEBITOS RECLAMAN CADA CREDITO
    // =====================================================
    ContarDebitosCredito = Table.AddColumn(
        CreditosIndexados,
        "TOTAL_CANDIDATOS_CREDITO",
        each
            let
                CreditoActual = _,

                DebitosCandidatos = Table.SelectRows(
                    ContarCandidatosDebito,
                    (Debito) =>
                        List.Contains(
                            Table.Column(
                                Debito[CANDIDATOS],
                                "ID_CREDITO"
                            ),
                            CreditoActual[ID_CREDITO]
                        )
                )
            in
                Table.RowCount(DebitosCandidatos),
        Int64.Type
    ),

    // =====================================================
    // 7. DETERMINAR SI LA PAREJA ES UNICA
    // =====================================================
    IdentificarCreditoUnico = Table.AddColumn(
        ContarCandidatosDebito,
        "DATOS_CREDITO",
        each
            if [TOTAL_CANDIDATOS_DEBITO] = 1 then
                let
                    IdCredito = [CANDIDATOS]{0}[ID_CREDITO],

                    CreditoEncontrado = Table.SelectRows(
                        ContarDebitosCredito,
                        each [ID_CREDITO] = IdCredito
                    )
                in
                    if
                        Table.RowCount(CreditoEncontrado) = 1
                        and CreditoEncontrado{0}[TOTAL_CANDIDATOS_CREDITO] = 1
                    then
                        CreditoEncontrado{0}
                    else
                        null
            else
                null
    ),

    // =====================================================
    // 8. CLASIFICAR LOS DEBITOS
    // =====================================================
    ClasificarDebitos = Table.AddColumn(
        IdentificarCreditoUnico,
        "ESTADO_CONCILIACION",
        each
            if [TOTAL_CANDIDATOS_DEBITO] = 0 then
                "SOLO_DEBITO"

            else if [TOTAL_CANDIDATOS_DEBITO] > 1 then
                "AMBIGUO_DEBITO"

            else if [DATOS_CREDITO] = null then
                "AMBIGUO_CREDITO"

            else
                "CONCILIADO",
        type text
    ),

    // =====================================================
    // 9. EXTRAER DATOS DEL CREDITO CONCILIADO
    // =====================================================
    AgregarFechaCredito = Table.AddColumn(
        ClasificarDebitos,
        "FECHA_CREDITO",
        each
            if [ESTADO_CONCILIACION] = "CONCILIADO" then
                [DATOS_CREDITO][#"FECHA DE MOV."]
            else
                null,
        type nullable datetime
    ),

    AgregarCodigoCredito = Table.AddColumn(
        AgregarFechaCredito,
        "CODIGO_CREDITO",
        each
            if [ESTADO_CONCILIACION] = "CONCILIADO" then
                [DATOS_CREDITO][#"CODIGO MOV."]
            else
                null
    ),

    AgregarCodigoDebito = Table.AddColumn(
        AgregarCodigoCredito,
        "CODIGO_DEBITO",
        each [#"CODIGO MOV."]
    ),

    AgregarFechaDebito = Table.AddColumn(
        AgregarCodigoDebito,
        "FECHA_DEBITO",
        each [#"FECHA DE MOV."],
        type datetime
    ),

    AgregarEmisor = Table.AddColumn(
        AgregarFechaDebito,
        "ID_EMISOR",
        each [ID_EMISOR_CARTOLA],
        type text
    ),

    AgregarBeneficiario = Table.AddColumn(
        AgregarEmisor,
        "ID_BENEFICIARIO",
        each [ID_CONTRAPARTE_ENDOSO],
        type text
    ),

    AgregarMonto = Table.AddColumn(
        AgregarBeneficiario,
        "MONTO_OPERACION",
        each [#"MONTO MOV."],
        type number
    ),

    AgregarDiferencia = Table.AddColumn(
        AgregarMonto,
        "DIFERENCIA_MINUTOS",
        each
            if [FECHA_CREDITO] = null then
                null
            else
                Number.Abs(
                    Duration.TotalMinutes(
                        [FECHA_CREDITO] - [FECHA_DEBITO]
                    )
                ),
        type nullable number
    ),

    AgregarCandidatosCredito = Table.AddColumn(
        AgregarDiferencia,
        "TOTAL_CANDIDATOS_CREDITO",
        each
            if [DATOS_CREDITO] = null then
                null
            else
                [DATOS_CREDITO][TOTAL_CANDIDATOS_CREDITO],
        Int64.Type
    ),

    // =====================================================
    // 10. RESULTADO DE LOS DEBITOS
    // =====================================================
    ResultadoDebitos = Table.SelectColumns(
        AgregarCandidatosCredito,
        {
            "ESTADO_CONCILIACION",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "MONTO_OPERACION",
            "FECHA_DEBITO",
            "FECHA_CREDITO",
            "DIFERENCIA_MINUTOS",
            "CODIGO_DEBITO",
            "CODIGO_CREDITO",
            "TOTAL_CANDIDATOS_DEBITO",
            "TOTAL_CANDIDATOS_CREDITO"
        }
    ),

    // =====================================================
    // 11. IDENTIFICAR CREDITOS SIN PAREJA CONFIRMADA
    // =====================================================
    CreditosSinConciliar = Table.SelectRows(
        ContarDebitosCredito,
        each [TOTAL_CANDIDATOS_CREDITO] <> 1
    ),

    ClasificarCreditos = Table.AddColumn(
        CreditosSinConciliar,
        "ESTADO_CONCILIACION",
        each
            if [TOTAL_CANDIDATOS_CREDITO] = 0 then
                "SOLO_CREDITO"
            else
                "AMBIGUO_CREDITO",
        type text
    ),

    CreditosAgregarEmisor = Table.AddColumn(
        ClasificarCreditos,
        "ID_EMISOR",
        each null,
        type nullable text
    ),

    CreditosAgregarBeneficiario = Table.AddColumn(
        CreditosAgregarEmisor,
        "ID_BENEFICIARIO",
        each [ID_BENEFICIARIO_CARTOLA],
        type text
    ),

    CreditosAgregarMonto = Table.AddColumn(
        CreditosAgregarBeneficiario,
        "MONTO_OPERACION",
        each [#"MONTO MOV."],
        type number
    ),

    CreditosAgregarFechaDebito = Table.AddColumn(
        CreditosAgregarMonto,
        "FECHA_DEBITO",
        each null,
        type nullable datetime
    ),

    CreditosAgregarFechaCredito = Table.AddColumn(
        CreditosAgregarFechaDebito,
        "FECHA_CREDITO",
        each [#"FECHA DE MOV."],
        type datetime
    ),

    CreditosAgregarDiferencia = Table.AddColumn(
        CreditosAgregarFechaCredito,
        "DIFERENCIA_MINUTOS",
        each null,
        type nullable number
    ),

    CreditosAgregarCodigoDebito = Table.AddColumn(
        CreditosAgregarDiferencia,
        "CODIGO_DEBITO",
        each null
    ),

    CreditosAgregarCodigoCredito = Table.AddColumn(
        CreditosAgregarCodigoDebito,
        "CODIGO_CREDITO",
        each [#"CODIGO MOV."]
    ),

    CreditosAgregarCandidatosDebito = Table.AddColumn(
        CreditosAgregarCodigoCredito,
        "TOTAL_CANDIDATOS_DEBITO",
        each null,
        Int64.Type
    ),

    ResultadoCreditos = Table.SelectColumns(
        CreditosAgregarCandidatosDebito,
        {
            "ESTADO_CONCILIACION",
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "MONTO_OPERACION",
            "FECHA_DEBITO",
            "FECHA_CREDITO",
            "DIFERENCIA_MINUTOS",
            "CODIGO_DEBITO",
            "CODIGO_CREDITO",
            "TOTAL_CANDIDATOS_DEBITO",
            "TOTAL_CANDIDATOS_CREDITO"
        }
    ),

    // =====================================================
    // 12. UNIFICAR RESULTADOS
    // =====================================================
    CombinarResultados = Table.Combine(
        {
            ResultadoDebitos,
            ResultadoCreditos
        }
    ),

    OrdenarResultados = Table.Sort(
        CombinarResultados,
        {
            {"ESTADO_CONCILIACION", Order.Ascending},
            {"ID_EMISOR", Order.Ascending},
            {"ID_BENEFICIARIO", Order.Ascending},
            {"FECHA_DEBITO", Order.Ascending}
        }
    )
in
    OrdenarResultados
