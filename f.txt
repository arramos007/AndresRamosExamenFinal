let
    Origen = qryBase,

    TiposTexto = {
        {"REL", type text},
        {"RELAC.", type text},
        {"RUC SOCIAL EMPLEADOR", type text},
        {"RAZON SOCIAL EMPLEADOR", type text},
        {"IDENTIFICACION EMPLEADO (EF)", type text},
        {"IDENT. EMPLEADO (EF) - 2", type text},
        {"NOMBRE EMPLEADO (EF)", type text},
        {"FALL.", type text},
        {"CODIGO MOV.", type text},
        {"CODIGO LIQUID.", type text},
        {"ACTO ADMINIS.", type text},
        {"EN TRAMA EMISOR", type text},
        {"IDENTIFICACION DEL EMISOR / ENDOSANTE", type text},
        {"NOMBRE DEL EMISOR / ENDOSANTE", type text},
        {"EN TRAMA BENEF.", type text},
        {"IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION", type text},
        {"NOMBRE DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION", type text},
        {"NOMBRE MOV.", type text},
        {"TIPO MOV.", type text},
        {"INST.", type text}
    },

    NormalizarTexto = Table.TransformColumnTypes(
        Origen,
        TiposTexto,
        "es-EC"
    ),

    TiposNumericos = Table.TransformColumnTypes(
        NormalizarTexto,
        {
            {"EDAD", Int64.Type},
            {"MONTO MOV.", type number},
            {"SALDO NCD", type number},
            {"FECHA DE MOV.", type datetime}
        },
        "es-EC"
    ),

    ClasificarMovimientos = Table.AddColumn(
        TiposNumericos,
        "CLASIFICACION_MOVIMIENTO",
        each
            let
                NombreMovimiento = Text.Upper(
                    Text.Trim(
                        if [#"NOMBRE MOV."] = null
                        then ""
                        else [#"NOMBRE MOV."]
                    )
                ),

                TipoMovimiento = Text.Upper(
                    Text.Trim(
                        if [#"TIPO MOV."] = null
                        then ""
                        else [#"TIPO MOV."]
                    )
                ),

                ActoAdministrativo = Text.Trim(
                    if [#"ACTO ADMINIS."] = null
                    then ""
                    else [#"ACTO ADMINIS."]
                )
            in
                if NombreMovimiento = "EMISION"
                    and TipoMovimiento = "CRE"
                    and ActoAdministrativo <> ""
                then "ACREDITACION_SRI"

                else if NombreMovimiento = "ENDOSO"
                    and TipoMovimiento = "CRE"
                then "ENDOSO_RECIBIDO"

                else if NombreMovimiento = "ENDOSO"
                    and TipoMovimiento = "DEB"
                then "ENDOSO_REALIZADO"

                else if Text.Contains(NombreMovimiento, "UTILIZ")
                then "UTILIZACION"

                else "OTROS",
        type text
    ),
ClasificarFlujo = Table.AddColumn(
        ClasificarMovimientos,
        "TIPO_FLUJO",
        each
            let
                Emisor = Text.Upper(
                    Text.Trim(
                        if [#"EN TRAMA EMISOR"] = null
                        then ""
                        else [#"EN TRAMA EMISOR"]
                    )
                ),
                Beneficiario = Text.Upper(
                    Text.Trim(
                        if [#"EN TRAMA BENEF."] = null
                        then ""
                        else [#"EN TRAMA BENEF."]
                    )
                )
            in
                if Emisor = "T" and Beneficiario = "T" then "INTERNO"
                else if Emisor = "T" and Beneficiario = "" then "SALIDA_TRAMA"
                else if Emisor = "" and Beneficiario = "T" then "ENTRADA_TRAMA"
                else if Emisor = "" and Beneficiario = "" then "EXTERNO"
                else "REVISAR_MARCAS",
        type text
    ),
    #"Personalizada agregada" = Table.AddColumn(ClasificarFlujo, "TIPO_OPERACION", each if [CLASIFICACION_MOVIMIENTO] = "ACREDITACION_SRI" then "ACREDITACION_ORIGEN_SRI"
else if [CLASIFICACION_MOVIMIENTO] = "ENDOSO_RECIBIDO" and [TIPO_FLUJO] = "INTERNO" then "RECEPCION_INTERNA"
else if [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO" and [TIPO_FLUJO] = "INTERNO" then "TRANSFERENCIA_INTERNA"
else if [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO" and [TIPO_FLUJO] = "SALIDA_TRAMA" then "SALIDA_A_EXTERNO"
else if [CLASIFICACION_MOVIMIENTO] = "ENDOSO_RECIBIDO" and [TIPO_FLUJO] = "ENTRADA_TRAMA" then "ENTRADA_DESDE_EXTERNO"
else if [CLASIFICACION_MOVIMIENTO] = "UTILIZACION" then "UTILIZACION"
else "OTROS"),
    #"Personalizada agregada1" = Table.AddColumn(#"Personalizada agregada", "ID_TITULAR", each if [#"IDENTIFICACION EMPLEADO (EF)"] = null 
then null else Text.Trim(Text.From([#"IDENTIFICACION EMPLEADO (EF)"]))),
    #"Personalizada agregada2" = Table.AddColumn(#"Personalizada agregada1", "ID_EMISOR", each if [#"IDENTIFICACION DEL EMISOR / ENDOSANTE"] = null then null else Text.Trim(Text.From([#"IDENTIFICACION DEL EMISOR / ENDOSANTE"]))),
    #"Personalizada agregada3" = Table.AddColumn(#"Personalizada agregada2", "ID_BENEFICIARIO", each if [#"IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"] = null then null else Text.Trim(Text.From([#"IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"]))),
    #"Personalizada agregada4" = Table.AddColumn(#"Personalizada agregada3", "ROL_TITULAR", each if [ID_TITULAR] = null or [ID_TITULAR] = "" then "SIN IDENTIFICAR" else if [ID_TITULAR] = [ID_EMISOR] and [ID_TITULAR] = [ID_BENEFICIARIO] then "EMISOR_Y_BENEFICIARIO" else if [ID_TITULAR] = [ID_EMISOR] then "EMISOR" else if [ID_TITULAR] = [ID_BENEFICIARIO] then "BENEFICIARIO" else "TERCERO_O_NO_COINCIDENTE"),
    #"Personalizada agregada5" = Table.AddColumn(#"Personalizada agregada4", "ID_CONTRAPARTE", each if [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO" and [ID_TITULAR] = [ID_EMISOR] then [ID_BENEFICIARIO] else if [CLASIFICACION_MOVIMIENTO] = "ENDOSO_RECIBIDO" and [ID_TITULAR] = [ID_BENEFICIARIO] then [ID_EMISOR] else null),
    #"Personalizada agregada6" = Table.AddColumn(#"Personalizada agregada5", "CLAVE_MOVIMIENTO", each Text.Combine(
    {
        if [#"CODIGO MOV."] = null then "" else Text.Trim([#"CODIGO MOV."]),
        if [ID_EMISOR] = null then "" else [ID_EMISOR],
        if [ID_BENEFICIARIO] = null then "" else [ID_BENEFICIARIO],
        if [#"FECHA DE MOV."] = null then ""
        else DateTime.ToText([#"FECHA DE MOV."], "yyyyMMddHHmmss"),
        if [#"MONTO MOV."] = null then ""
        else Number.ToText([#"MONTO MOV."], "0.00", "en-US")
    },
    "|"
)),
    #"Personalizada agregada7" = Table.AddColumn(#"Personalizada agregada6", "ESTADO_TRAZABILIDAD", each if [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO" then
    if [ID_TITULAR] = null or [ID_TITULAR] = "" then "SIN_TITULAR"
    else if [ID_EMISOR] <> [ID_TITULAR] then "EMISOR_NO_COINCIDE_CON_TITULAR"
    else if [ID_BENEFICIARIO] = null or [ID_BENEFICIARIO] = "" then "SIN_BENEFICIARIO"
    else "TRAZABILIDAD_COMPLETA"

else if [CLASIFICACION_MOVIMIENTO] = "ENDOSO_RECIBIDO" then
    if [ID_TITULAR] = null or [ID_TITULAR] = "" then "SIN_TITULAR"
    else if [ID_BENEFICIARIO] <> [ID_TITULAR] then "BENEFICIARIO_NO_COINCIDE_CON_TITULAR"
    else if [ID_EMISOR] = null or [ID_EMISOR] = "" then "SIN_EMISOR"
    else "TRAZABILIDAD_COMPLETA"

else "NO_APLICA")
in
    #"Personalizada agregada7"
