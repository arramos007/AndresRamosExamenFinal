let
    Origen = qryEndososEconomicos,

    AgruparEndosos = Table.Group(
        Origen,
        {
            "ID_EMISOR",
            "ID_BENEFICIARIO",
            "TIPO_OPERACION"
        },
        {
            {
                "TOTAL_OPERACIONES",
                each Table.RowCount(_),
                Int64.Type
            },
            {
                "MONTO_TOTAL",
                each List.Sum(List.RemoveNulls([MONTO_OPERACION])),
                type number
            },
            {
                "FECHA_PRIMERA_OPERACION",
                each List.Min([FECHA_OPERACION]),
                type datetime
            },
            {
                "FECHA_ULTIMA_OPERACION",
                each List.Max([FECHA_OPERACION]),
                type datetime
            }
        }
    ),

    OrdenarResultados = Table.Sort(
        AgruparEndosos,
        {
            {"MONTO_TOTAL", Order.Descending}
        }
    )
in
    OrdenarResultados
