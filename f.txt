qry37_ValidacionSegundoSalto


let
    Origen =
        qry36_ConexionesCronologicasSegundoSalto,

    TotalSecuencias =
        Table.RowCount(Origen),

    ParesOperacionesDistintas =
        Table.RowCount(
            Table.Distinct(
                Table.SelectColumns(
                    Origen,
                    {
                        "ID_REGISTRO_DEBITO",
                        "ID_DEBITO_SEGUNDO_SALTO"
                    }
                )
            )
        ),

    ParesDuplicados =
        TotalSecuencias - ParesOperacionesDistintas,

    OrigenesInconsistentes =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each
                    [ID_PARTICIPANTE_DESTINO]
                    <> [ORIGEN_SEGUNDO_SALTO]
            )
        ),

    SecuenciasConTiempoNegativo =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [SEGUNDOS_ENTRE_OPERACIONES] < 0
            )
        ),

    SecuenciasTemporalesAmbiguas =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each
                    [ESTADO_SECUENCIA]
                    = "ORDEN_TEMPORAL_AMBIGUO"
            )
        ),

    SecuenciasPosteriores =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each
                    [ESTADO_SECUENCIA]
                    = "CRONOLOGICAMENTE_POSTERIOR"
            )
        ),

    CiclosDetectados =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ES_CICLO_SECUENCIA] = true
            )
        ),

    Resultado =
        #table(
            {
                "INDICADOR",
                "CANTIDAD"
            },
            {
                {"TOTAL_SECUENCIAS", TotalSecuencias},
                {"PARES_OPERACIONES_DISTINTAS", ParesOperacionesDistintas},
                {"PARES_DUPLICADOS", ParesDuplicados},
                {"ORIGENES_INCONSISTENTES", OrigenesInconsistentes},
                {"SECUENCIAS_CON_TIEMPO_NEGATIVO", SecuenciasConTiempoNegativo},
                {"SECUENCIAS_TEMPORALES_AMBIGUAS", SecuenciasTemporalesAmbiguas},
                {"SECUENCIAS_CRONOLOGICAMENTE_POSTERIORES", SecuenciasPosteriores},
                {"CICLOS_DETECTADOS", CiclosDetectados}
            }
        )
in
    Resultado
