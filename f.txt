
let
    Origen = qry13_ConciliacionEndosos,

    AgruparEstados =
        Table.Group(
            Origen,
            {"ESTADO_CONCILIACION"},
            {
                {
                    "CANTIDAD_FILAS",
                    each Table.RowCount(_),
                    Int64.Type
                },
                {
                    "DEBITOS_DISTINTOS",
                    each
                        Table.RowCount(
                            Table.Distinct(
                                Table.SelectRows(
                                    Table.SelectColumns(
                                        _,
                                        {
                                            "ID_REGISTRO_DEBITO"
                                        }
                                    ),
                                    each [ID_REGISTRO_DEBITO] <> null
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "CREDITOS_DISTINTOS",
                    each
                        Table.RowCount(
                            Table.Distinct(
                                Table.SelectRows(
                                    Table.SelectColumns(
                                        _,
                                        {
                                            "ID_REGISTRO_CREDITO"
                                        }
                                    ),
                                    each [ID_REGISTRO_CREDITO] <> null
                                )
                            )
                        ),
                    Int64.Type
                },
                {
                    "IMPORTE_DEBITOS",
                    each
                        let
                            Registros =
                                Table.Distinct(
                                    Table.SelectRows(
                                        Table.SelectColumns(
                                            _,
                                            {
                                                "ID_REGISTRO_DEBITO",
                                                "MONTO_DEBITO"
                                            }
                                        ),
                                        each
                                            [ID_REGISTRO_DEBITO] <> null
                                    ),
                                    {"ID_REGISTRO_DEBITO"}
                                )
                        in
                            if Table.IsEmpty(Registros) then
                                null
                            else
                                List.Sum(
                                    Registros[MONTO_DEBITO]
                                ),
                    type nullable number
                },
                {
                    "IMPORTE_CREDITOS",
                    each
                        let
                            Registros =
                                Table.Distinct(
                                    Table.SelectRows(
                                        Table.SelectColumns(
                                            _,
                                            {
                                                "ID_REGISTRO_CREDITO",
                                                "MONTO_CREDITO"
                                            }
                                        ),
                                        each
                                            [ID_REGISTRO_CREDITO] <> null
                                    ),
                                    {"ID_REGISTRO_CREDITO"}
                                )
                        in
                            if Table.IsEmpty(Registros) then
                                null
                            else
                                List.Sum(
                                    Registros[MONTO_CREDITO]
                                ),
                    type nullable number
                }
            }
        ),

    Resultado =
        Table.Sort(
            AgruparEstados,
            {
                {"ESTADO_CONCILIACION", Order.Ascending}
            }
        )
in
    Resultado
