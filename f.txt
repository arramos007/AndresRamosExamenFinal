let
    Origen = qryDetalleBeneficiariosOrigen,

    AgruparDatos = Table.Group(
        Origen,
        {
            "ID_BENEFICIARIO",
            "PERFIL_ORIGEN"
        },
        {
            {
                "TOTAL_MOVIMIENTOS",
                each Table.RowCount(_),
                Int64.Type
            },
            {
                "TOTAL_PARTICIPANTES_ORIGEN",
                each List.Count(
                    List.Distinct(
                        List.RemoveNulls(
                            Table.Column(_, "ID_TITULAR")
                        )
                    )
                ),
                Int64.Type
            },
            {
                "MONTO_TOTAL",
                each List.Sum(
                    List.RemoveNulls(
                        Table.Column(_, "MONTO MOV.")
                    )
                ),
                type number
            },
            {
                "FECHA_PRIMER_MOVIMIENTO",
                each List.Min(
                    Table.Column(_, "FECHA DE MOV.")
                ),
                type datetime
            },
            {
                "FECHA_ULTIMO_MOVIMIENTO",
                each List.Max(
                    Table.Column(_, "FECHA DE MOV.")
                ),
                type datetime
            }
        }
    ),

    OrdenarResultados = Table.Sort(
        AgruparDatos,
        {
            {"ID_BENEFICIARIO", Order.Ascending},
            {"MONTO_TOTAL", Order.Descending}
        }
    )
in
    OrdenarResultados
