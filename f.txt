
let
    // 1. Preparar débitos
    DebitosOrigen = qry08_EndososDebito,

    DebitosBase =
        Table.SelectColumns(
            DebitosOrigen,
            {
                "ID_REGISTRO_ORIGEN",
                "IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION",
                "FECHA DE MOV.",
                "MONTO MOV."
            }
        ),

    DebitosRenombrados =
        Table.RenameColumns(
            DebitosBase,
            {
                {"ID_REGISTRO_ORIGEN", "ID_REGISTRO_DEBITO"},
                {
                    "IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION",
                    "ID_BENEFICIARIO_DEBITO"
                },
                {"FECHA DE MOV.", "FECHA_DEBITO"},
                {"MONTO MOV.", "MONTO_DEBITO"}
            }
        ),

    Debitos =
        Table.AddColumn(
            DebitosRenombrados,
            "ID_COMPARACION",
            each
                if [ID_BENEFICIARIO_DEBITO] = null then
                    null
                else
                    Text.Trim(
                        Text.From([ID_BENEFICIARIO_DEBITO])
                    ),
            type nullable text
        ),

    // 2. Preparar créditos
    CreditosOrigen = qry09_EndososCredito,

    CreditosBase =
        Table.SelectColumns(
            CreditosOrigen,
            {
                "ID_REGISTRO_ORIGEN",
                "IDENTIFICACION EMPLEADO (EF)",
                "FECHA DE MOV.",
                "MONTO MOV."
            }
        ),

    CreditosRenombrados =
        Table.RenameColumns(
            CreditosBase,
            {
                {"ID_REGISTRO_ORIGEN", "ID_REGISTRO_CREDITO"},
                {
                    "IDENTIFICACION EMPLEADO (EF)",
                    "ID_TITULAR_CREDITO"
                },
                {"FECHA DE MOV.", "FECHA_CREDITO"},
                {"MONTO MOV.", "MONTO_CREDITO"}
            }
        ),

    Creditos =
        Table.AddColumn(
            CreditosRenombrados,
            "ID_COMPARACION",
            each
                if [ID_TITULAR_CREDITO] = null then
                    null
                else
                    Text.Trim(
                        Text.From([ID_TITULAR_CREDITO])
                    ),
            type nullable text
        ),

    // 3. Agrupar créditos por identificación e importe
    CreditosValidos =
        Table.SelectRows(
            Creditos,
            each
                [ID_COMPARACION] <> null
                and [MONTO_CREDITO] <> null
                and [FECHA_CREDITO] <> null
        ),

    AgruparCreditos =
        Table.Group(
            CreditosValidos,
            {"ID_COMPARACION", "MONTO_CREDITO"},
            {
                {
                    "REGISTROS_CREDITO",
                    each Table.SelectColumns(
                        _,
                        {
                            "ID_REGISTRO_CREDITO",
                            "ID_TITULAR_CREDITO",
                            "FECHA_CREDITO"
                        }
                    ),
                    type table
                }
            }
        ),

    // 4. Buscar grupos correspondientes a cada débito
    UnirGrupos =
        Table.NestedJoin(
            Debitos,
            {
                "ID_COMPARACION",
                "MONTO_DEBITO"
            },
            AgruparCreditos,
            {
                "ID_COMPARACION",
                "MONTO_CREDITO"
            },
            "GRUPO_CREDITOS",
            JoinKind.LeftOuter
        ),

    // 5. Filtrar por diferencia máxima de tres segundos
    BuscarCoincidencias =
        Table.AddColumn(
            UnirGrupos,
            "CREDITOS_CANDIDATOS",
            each
                let
                    FechaDebito = [FECHA_DEBITO],

                    Grupos =
                        [GRUPO_CREDITOS],

                    Registros =
                        if Table.IsEmpty(Grupos) then
                            #table(
                                {
                                    "ID_REGISTRO_CREDITO",
                                    "ID_TITULAR_CREDITO",
                                    "FECHA_CREDITO"
                                },
                                {}
                            )
                        else
                            Grupos{0}[REGISTROS_CREDITO],

                    Coincidencias =
                        if FechaDebito = null then
                            Table.FirstN(Registros, 0)
                        else
                            Table.SelectRows(
                                Registros,
                                (Credito) =>
                                    Number.Abs(
                                        Duration.TotalSeconds(
                                            Credito[FECHA_CREDITO]
                                            - FechaDebito
                                        )
                                    ) <= 3
                            ),

                    AgregarDiferencia =
                        Table.AddColumn(
                            Coincidencias,
                            "DIFERENCIA_SEGUNDOS",
                            each Number.Abs(
                                Duration.TotalSeconds(
                                    [FECHA_CREDITO]
                                    - FechaDebito
                                )
                            ),
                            type number
                        )
                in
                    AgregarDiferencia,
            type table
        ),

    // 6. Contar candidatos
    ContarCandidatos =
        Table.AddColumn(
            BuscarCoincidencias,
            "CANTIDAD_CANDIDATOS_DEBITO",
            each Table.RowCount([CREDITOS_CANDIDATOS]),
            Int64.Type
        ),

    // 7. Expandir únicamente los candidatos
    ExpandirCandidatos =
        Table.ExpandTableColumn(
            ContarCandidatos,
            "CREDITOS_CANDIDATOS",
            {
                "ID_REGISTRO_CREDITO",
                "ID_TITULAR_CREDITO",
                "FECHA_CREDITO",
                "DIFERENCIA_SEGUNDOS"
            },
            {
                "ID_REGISTRO_CREDITO",
                "ID_TITULAR_CREDITO",
                "FECHA_CREDITO",
                "DIFERENCIA_SEGUNDOS"
            }
        ),

    LimpiarGrupos =
        Table.RemoveColumns(
            ExpandirCandidatos,
            {"GRUPO_CREDITOS", "ID_COMPARACION"}
        ),

    // 8. Clasificar resultados de débitos
    ClasificarDebitos =
        Table.AddColumn(
            LimpiarGrupos,
            "TIPO_REGISTRO",
            each
                if [CANTIDAD_CANDIDATOS_DEBITO] = 0 then
                    "DEBITO_SIN_CANDIDATO"
                else
                    "CANDIDATO",
            type text
        ),

    // 9. Identificar créditos con al menos un candidato
    IDsCreditosCandidatos =
        Table.Distinct(
            Table.SelectColumns(
                Table.SelectRows(
                    ClasificarDebitos,
                    each [ID_REGISTRO_CREDITO] <> null
                ),
                {"ID_REGISTRO_CREDITO"}
            )
        ),

    CreditosSinCandidato =
        Table.NestedJoin(
            Creditos,
            {"ID_REGISTRO_CREDITO"},
            IDsCreditosCandidatos,
            {"ID_REGISTRO_CREDITO"},
            "COINCIDENCIA",
            JoinKind.LeftAnti
        ),

    CreditosPendientesBase =
        Table.RemoveColumns(
            CreditosSinCandidato,
            {"COINCIDENCIA", "ID_COMPARACION"}
        ),

    // 10. Completar estructura de créditos pendientes
    CreditosPendientes1 =
        Table.AddColumn(
            CreditosPendientesBase,
            "ID_REGISTRO_DEBITO",
            each null,
            type nullable number
        ),

    CreditosPendientes2 =
        Table.AddColumn(
            CreditosPendientes1,
            "ID_BENEFICIARIO_DEBITO",
            each null,
            type nullable text
        ),

    CreditosPendientes3 =
        Table.AddColumn(
            CreditosPendientes2,
            "FECHA_DEBITO",
            each null,
            type nullable datetime
        ),

    CreditosPendientes4 =
        Table.AddColumn(
            CreditosPendientes3,
            "MONTO_DEBITO",
            each null,
            type nullable number
        ),

    CreditosPendientes5 =
        Table.AddColumn(
            CreditosPendientes4,
            "DIFERENCIA_SEGUNDOS",
            each null,
            type nullable number
        ),

    CreditosPendientes6 =
        Table.AddColumn(
            CreditosPendientes5,
            "CANTIDAD_CANDIDATOS_DEBITO",
            each 0,
            Int64.Type
        ),

    CreditosPendientes7 =
        Table.AddColumn(
            CreditosPendientes6,
            "TIPO_REGISTRO",
            each "CREDITO_SIN_CANDIDATO",
            type text
        ),

    // 11. Unificar resultados
    Resultados =
        Table.Combine(
            {
                ClasificarDebitos,
                CreditosPendientes7
            }
        ),

    SeleccionarColumnas =
        Table.SelectColumns(
            Resultados,
            {
                "TIPO_REGISTRO",
                "ID_REGISTRO_DEBITO",
                "ID_REGISTRO_CREDITO",
                "ID_BENEFICIARIO_DEBITO",
                "ID_TITULAR_CREDITO",
                "FECHA_DEBITO",
                "FECHA_CREDITO",
                "MONTO_DEBITO",
                "MONTO_CREDITO",
                "DIFERENCIA_SEGUNDOS",
                "CANTIDAD_CANDIDATOS_DEBITO"
            }
        )
in
    SeleccionarColumnas
