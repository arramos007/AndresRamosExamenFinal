qry27_DetallePerfilesEntidades


let
    Origen =
        qry25_PerfilesPreliminaresEntidades,

    SeleccionarColumnas =
        Table.SelectColumns(
            Origen,
            {
                "RUC_ENTIDAD",
                "RAZON_SOCIAL",
                "TIPO_ENTIDAD",
                "PRESENCIA_MOVIMIENTOS",
                "CANTIDAD_ENDOSOS_RECIBIDOS",
                "TOTAL_RECIBIDO",
                "CANTIDAD_RUC_EMISORES_RELACIONADOS",
                "CANTIDAD_ENDOSOS_REALIZADOS",
                "TOTAL_REALIZADO",
                "CANTIDAD_RUC_BENEFICIARIOS_RELACIONADOS",
                "DIFERENCIA_RECIBIDO_REALIZADO",
                "RATIO_SALIDA_ENTRADA",
                "POSIBLE_CONCENTRADORA",
                "POSIBLE_ACUMULADORA",
                "POSIBLE_INTERMEDIARIA",
                "POSIBLE_REDISTRIBUIDORA",
                "CANTIDAD_PERFILES",
                "OBSERVACIONES_PERFIL"
            }
        ),

    OrdenarResultados =
        Table.Sort(
            SeleccionarColumnas,
            {
                {"CANTIDAD_PERFILES", Order.Descending},
                {"TOTAL_RECIBIDO", Order.Descending}
            }
        ),

    AgregarPosicion =
        Table.AddIndexColumn(
            OrdenarResultados,
            "POSICION_ANALISIS",
            1,
            1,
            Int64.Type
        ),

    Resultado =
        Table.ReorderColumns(
            AgregarPosicion,
            {
                "POSICION_ANALISIS",
                "RUC_ENTIDAD",
                "RAZON_SOCIAL",
                "TIPO_ENTIDAD",
                "PRESENCIA_MOVIMIENTOS",
                "TOTAL_RECIBIDO",
                "TOTAL_REALIZADO",
                "DIFERENCIA_RECIBIDO_REALIZADO",
                "RATIO_SALIDA_ENTRADA",
                "CANTIDAD_ENDOSOS_RECIBIDOS",
                "CANTIDAD_ENDOSOS_REALIZADOS",
                "CANTIDAD_RUC_EMISORES_RELACIONADOS",
                "CANTIDAD_RUC_BENEFICIARIOS_RELACIONADOS",
                "POSIBLE_CONCENTRADORA",
                "POSIBLE_ACUMULADORA",
                "POSIBLE_INTERMEDIARIA",
                "POSIBLE_REDISTRIBUIDORA",
                "CANTIDAD_PERFILES",
                "OBSERVACIONES_PERFIL"
            }
        )
in
    Resultado
