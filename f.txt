let
    Origen = qryMovimientosClasificados,

    Filtrar = Table.SelectRows(
        Origen,
        each [#"CODIGO MOV."] = 45493225
    ),

    Seleccionar = Table.SelectColumns(
        Filtrar,
        {
            "IDENTIFICACION EMPLEADO (EF)",
            "RUC SOCIAL EMPLEADOR",
            "CODIGO MOV.",
            "CODIGO LIQUID.",
            "ACTO ADMINIS.",
            "FECHA DE MOV.",
            "IDENTIFICACION DEL EMISOR / ENDOSANTE",
            "IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION",
            "NOMBRE MOV.",
            "TIPO MOV.",
            "MONTO MOV.",
            "SALDO NCD",
            "CLASIFICACION_MOVIMIENTO"
        }
    ),

    Ordenar = Table.Sort(
        Seleccionar,
        {
            {"RUC SOCIAL EMPLEADOR", Order.Ascending}
        }
    )
in
    Ordenar
