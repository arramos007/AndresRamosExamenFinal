
let
    Origen =
        qry32_ConexionesParaRutas,

    Indice =
        qry34_IndiceConexionesPorParticipante,

    UnirSiguientesConexiones =
        Table.NestedJoin(
            Origen,
            {"ID_PARTICIPANTE_DESTINO"},
            Indice,
            {"ID_PARTICIPANTE_ORIGEN"},
            "SALIDAS_SIGUIENTES",
            JoinKind.Inner
        ),

    ExpandirTablas =
        Table.ExpandTableColumn(
            UnirSiguientesConexiones,
            "SALIDAS_SIGUIENTES",
            {"CONEXIONES_SALIDA"},
            {"TABLA_SIGUIENTES_SALIDAS"}
        ),

    ExpandirConexiones =
        Table.ExpandTableColumn(
            ExpandirTablas,
            "TABLA_SIGUIENTES_SALIDAS",
            {
                "ID_RUTA",
                "ID_REGISTRO_DEBITO",
                "ID_REGISTRO_CREDITO",
                "ID_PARTICIPANTE_ORIGEN",
                "ID_PARTICIPANTE_DESTINO",
                "FECHA_DEBITO",
                "FECHA_CREDITO",
                "MONTO_DEBITO",
                "MONTO_CREDITO",
                "ES_CICLO",
                "NIVEL_EVIDENCIA"
            },
            {
                "ID_RUTA_SEGUNDO_SALTO",
                "ID_DEBITO_SEGUNDO_SALTO",
                "ID_CREDITO_SEGUNDO_SALTO",
                "ORIGEN_SEGUNDO_SALTO",
                "DESTINO_SEGUNDO_SALTO",
                "FECHA_DEBITO_SEGUNDO_SALTO",
                "FECHA_CREDITO_SEGUNDO_SALTO",
                "MONTO_DEBITO_SEGUNDO_SALTO",
                "MONTO_CREDITO_SEGUNDO_SALTO",
                "ES_CICLO_SEGUNDO_SALTO",
                "NIVEL_EVIDENCIA_SEGUNDO_SALTO"
            }
        ),

    RenombrarEvidenciaPrimerSalto =
        Table.RenameColumns(
            ExpandirConexiones,
            {
                {
                    "NIVEL_EVIDENCIA",
                    "NIVEL_EVIDENCIA_PRIMER_SALTO"
                }
            }
        ),

    FiltrarSecuenciasTemporales =
        Table.SelectRows(
            RenombrarEvidenciaPrimerSalto,
            each
                [FECHA_DEBITO_SEGUNDO_SALTO]
                    >= [FECHA_CREDITO]
        ),

    AgregarIDRuta =
        Table.AddColumn(
            FiltrarSecuenciasTemporales,
            "ID_RUTA_DOS_SALTOS",
            each
                Text.From([ID_REGISTRO_DEBITO])
                & ">"
                & Text.From([ID_DEBITO_SEGUNDO_SALTO]),
            type text
        ),

    AgregarDiferenciaTemporal =
        Table.AddColumn(
            AgregarIDRuta,
            "SEGUNDOS_ENTRE_OPERACIONES",
            each
                Duration.TotalSeconds(
                    [FECHA_DEBITO_SEGUNDO_SALTO]
                    - [FECHA_CREDITO]
                ),
            type number
        ),

    AgregarEstadoTemporal =
        Table.AddColumn(
            AgregarDiferenciaTemporal,
            "ESTADO_SECUENCIA",
            each
                if [SEGUNDOS_ENTRE_OPERACIONES] = 0 then
                    "ORDEN_TEMPORAL_AMBIGUO"
                else
                    "CRONOLOGICAMENTE_POSTERIOR",
            type text
        ),

    AgregarCiclo =
        Table.AddColumn(
            AgregarEstadoTemporal,
            "ES_CICLO_SECUENCIA",
            each
                [ID_PARTICIPANTE_ORIGEN]
                    = [DESTINO_SEGUNDO_SALTO]
                or [ES_CICLO]
                or [ES_CICLO_SEGUNDO_SALTO],
            type logical
        ),

    AgregarEvidenciaSecuencia =
        Table.AddColumn(
            AgregarCiclo,
            "NIVEL_EVIDENCIA_SECUENCIA",
            each
                if [ESTADO_SECUENCIA] = "ORDEN_TEMPORAL_AMBIGUO" then
                    "CONEXION_OBSERVADA"
                else
                    "SECUENCIA_CONFIRMADA",
            type text
        ),

    SeleccionarResultado =
        Table.SelectColumns(
            AgregarEvidenciaSecuencia,
            {
                "ID_RUTA_DOS_SALTOS",
                "ID_REGISTRO_DEBITO",
                "ID_REGISTRO_CREDITO",
                "ID_PARTICIPANTE_ORIGEN",
                "ID_PARTICIPANTE_DESTINO",
                "FECHA_DEBITO",
                "FECHA_CREDITO",
                "MONTO_DEBITO",
                "NIVEL_EVIDENCIA_PRIMER_SALTO",
                "ES_CICLO",
                "ID_DEBITO_SEGUNDO_SALTO",
                "ID_CREDITO_SEGUNDO_SALTO",
                "ORIGEN_SEGUNDO_SALTO",
                "DESTINO_SEGUNDO_SALTO",
                "FECHA_DEBITO_SEGUNDO_SALTO",
                "FECHA_CREDITO_SEGUNDO_SALTO",
                "MONTO_DEBITO_SEGUNDO_SALTO",
                "MONTO_CREDITO_SEGUNDO_SALTO",
                "NIVEL_EVIDENCIA_SEGUNDO_SALTO",
                "ES_CICLO_SEGUNDO_SALTO",
                "SEGUNDOS_ENTRE_OPERACIONES",
                "ESTADO_SECUENCIA",
                "ES_CICLO_SECUENCIA",
                "NIVEL_EVIDENCIA_SECUENCIA"
            }
        ),

    Resultado =
        Table.Sort(
            SeleccionarResultado,
            {
                {"FECHA_DEBITO", Order.Ascending},
                {"ID_REGISTRO_DEBITO", Order.Ascending},
                {"FECHA_DEBITO_SEGUNDO_SALTO", Order.Ascending}
            }
        )
in
    Resultado
