
let
    Origen = qryRelacionesParticipantes,

    AgruparRelaciones = Table.Group(
        Origen,
        {
            "ID_EMISOR",
            "ID_BENEFICIARIO"
        },
        {
            {
                "TOTAL_MOVIMIENTOS",
                each Table.RowCount(_),
                Int64.Type
            },
            {
                "MONTO_TOTAL",
                each List.Sum(
                    List.Transform(
                        [#"MONTO MOV."],
                        each try Number.From(_) otherwise 0
                    )
                ),
                type number
            },
            {
                "FECHA_PRIMER_MOVIMIENTO",
                each List.Min([#"FECHA DE MOV."]),
                type datetime
            },
            {
                "FECHA_ULTIMO_MOVIMIENTO",
                each List.Max([#"FECHA DE MOV."]),
                type datetime
            }
        }
    ),

    OrdenarRelaciones = Table.Sort(
        AgruparRelaciones,
        {
            {"MONTO_TOTAL", Order.Descending}
        }
    )
in
    OrdenarRelaciones
