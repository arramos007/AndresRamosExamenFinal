    ClasificarFlujo = Table.AddColumn(
        ClasificarMovimientos,
        "TIPO_FLUJO",
        each
            let
                Emisor = Text.Upper(
                    Text.Trim(
                        if [#"EN TRAMA EMISOR"] = null
                        then ""
                        else [#"EN TRAMA EMISOR"]
                    )
                ),
                Beneficiario = Text.Upper(
                    Text.Trim(
                        if [#"EN TRAMA BENEF."] = null
                        then ""
                        else [#"EN TRAMA BENEF."]
                    )
                )
            in
                if Emisor = "T" and Beneficiario = "T" then "INTERNO"
                else if Emisor = "T" and Beneficiario = "" then "SALIDA_TRAMA"
                else if Emisor = "" and Beneficiario = "T" then "ENTRADA_TRAMA"
                else if Emisor = "" and Beneficiario = "" then "EXTERNO"
                else "REVISAR_MARCAS",
        type text
    )
in
    ClasificarFlujo
