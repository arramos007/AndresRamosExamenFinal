let
    Origen = qryAuditoriaDuplicadosExactos,

    CodigosUnicos = List.Distinct(
        Table.Column(Origen, "CODIGO MOV.")
    ),

    CodigosRepetidos = Table.SelectRows(
        Origen,
        each [TOTAL_REPETICIONES] > 1
    ),

    Propietarios = List.Distinct(
        List.RemoveNulls(
            Table.Column(Origen, "IDENTIFICACION EMPLEADO (EF)")
        )
    ),

    Empresas = List.Distinct(
        List.RemoveNulls(
            Table.Column(Origen, "RUC SOCIAL EMPLEADOR")
        )
    ),

    TotalRegistros = Table.RowCount(Origen),

    TotalRepeticiones = List.Sum(
        List.Transform(
            Table.Column(Origen, "TOTAL_REPETICIONES"),
            each if _ = null then 0 else _ - 1
        )
    ),

    Resultado = #table(
        {"INDICADOR", "VALOR"},
        {
            {"CODIGOS_AUDITADOS", List.Count(CodigosUnicos)},
            {"CODIGOS_CON_REPETICIONES", List.Count(
                List.Distinct(
                    Table.Column(CodigosRepetidos, "CODIGO MOV.")
                )
            )},
            {"REGISTROS_AUDITADOS", TotalRegistros},
            {"PROPIETARIOS_INVOLUCRADOS", List.Count(Propietarios)},
            {"EMPRESAS_INVOLUCRADAS", List.Count(Empresas)},
            {"REPETICIONES_ADICIONALES", TotalRepeticiones}
        }
    )
in
    Resultado
