let
    Origen = qryAuditoriaCodigoMov,

    Agrupar = Table.Group(
        Origen,
        {"CODIGO MOV."},
        {
            {"TOTAL_REGISTROS", each Table.RowCount(_), Int64.Type},

            {"RUC_PROPIETARIOS", each List.Count(
                List.Distinct(Table.Column(_, "IDENTIFICACION EMPLEADO (EF)"))
            ), Int64.Type},

            {"FECHAS_DISTINTAS", each List.Count(
                List.Distinct(Table.Column(_, "FECHA DE MOV."))
            ), Int64.Type},

            {"EMISORES_DISTINTOS", each List.Count(
                List.Distinct(Table.Column(_, "IDENTIFICACION DEL EMISOR / ENDOSANTE"))
            ), Int64.Type},

            {"BENEFICIARIOS_DISTINTOS", each List.Count(
                List.Distinct(Table.Column(_, "IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"))
            ), Int64.Type},

            {"MONTOS_DISTINTOS", each List.Count(
                List.Distinct(Table.Column(_, "MONTO MOV."))
            ), Int64.Type},

            {"TIPOS_MOVIMIENTO", each List.Count(
                List.Distinct(Table.Column(_, "TIPO MOV."))
            ), Int64.Type},

            {"CLASIFICACIONES", each List.Count(
                List.Distinct(Table.Column(_, "CLASIFICACION_MOVIMIENTO"))
            ), Int64.Type}
        }
    )
in
    Agrupar
