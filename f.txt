    Entradas = try Number.From([TOTAL_ENTRADAS_INTERNAS]) otherwise 0,
    Salidas = try Number.From([TOTAL_SALIDAS_EXTERNAS]) otherwise 0,
    Secuencia = [RESULTADO_SECUENCIA]
in
    if Entradas > 0 and Salidas > 0 and Secuencia = "ENTRADA_ANTERIOR_A_SALIDA" then
        "CONCENTRADOR + ACUMULADOR_COMPATIBLE"
    else if Entradas > 0 and Salidas > 0 then
        "CONCENTRADOR + SALIDA_EXTERNA (SECUENCIA NO CONFIRMADA)"
    else if Entradas > 0 then
        "CONCENTRADOR"
    else if Salidas > 0 then
        "EMISOR_SALIDA_EXTERNA"
    else
        "SIN_PERFIL_DE_RED_IDENTIFICADO"
