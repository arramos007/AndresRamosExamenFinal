qry29_ValidacionRutasDirectas


let
    Origen =
        qry28_RutasEconomicasDirectas,

    TotalFilas =
        Table.RowCount(Origen),

    DebitosDistintos =
        List.Count(
            List.Distinct(
                List.RemoveNulls(
                    Origen[ID_REGISTRO_DEBITO]
                )
            )
        ),

    CreditosDistintos =
        List.Count(
            List.Distinct(
                List.RemoveNulls(
                    Origen[ID_REGISTRO_CREDITO]
                )
            )
        ),

    DebitosDuplicados =
        TotalFilas - DebitosDistintos,

    CreditosDuplicados =
        TotalFilas - CreditosDistintos,

    ImportesIguales =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [MONTO_DEBITO] = [MONTO_CREDITO]
            )
        ),

    OrigenIdentificado =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ID_PARTICIPANTE_ORIGEN] <> null
            )
        ),

    DestinoIdentificado =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ID_PARTICIPANTE_DESTINO] <> null
            )
        ),

    Autorreferencias =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ES_CICLO] = true
            )
        ),

    Validaciones = {
        [
            VALIDACION = "TOTAL_OPERACIONES",
            RESULTADO = TotalFilas,
            ESPERADO = 1338
        ],
        [
            VALIDACION = "DEBITOS_DISTINTOS",
            RESULTADO = DebitosDistintos,
            ESPERADO = TotalFilas
        ],
        [
            VALIDACION = "CREDITOS_DISTINTOS",
            RESULTADO = CreditosDistintos,
            ESPERADO = TotalFilas
        ],
        [
            VALIDACION = "DEBITOS_DUPLICADOS",
            RESULTADO = DebitosDuplicados,
            ESPERADO = 0
        ],
        [
            VALIDACION = "CREDITOS_DUPLICADOS",
            RESULTADO = CreditosDuplicados,
            ESPERADO = 0
        ],
        [
            VALIDACION = "IMPORTES_IGUALES",
            RESULTADO = ImportesIguales,
            ESPERADO = TotalFilas
        ],
        [
            VALIDACION = "ORIGEN_IDENTIFICADO",
            RESULTADO = OrigenIdentificado,
            ESPERADO = TotalFilas
        ],
        [
            VALIDACION = "DESTINO_IDENTIFICADO",
            RESULTADO = DestinoIdentificado,
            ESPERADO = TotalFilas
        }
    },

    TablaValidaciones =
        Table.FromRecords(Validaciones),

    AgregarEstado =
        Table.AddColumn(
            TablaValidaciones,
            "ESTADO",
            each
                if [RESULTADO] = [ESPERADO] then
                    "CORRECTO"
                else
                    "REVISAR",
            type text
        )
in
    AgregarEstado
