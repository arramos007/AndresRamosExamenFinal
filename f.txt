qry23_DistribucionIndicadoresEntidades


let
    Origen = qry22_ComparacionEntidades,

    Indicadores = {
        "TOTAL_RECIBIDO",
        "TOTAL_REALIZADO",
        "DIFERENCIA_RECIBIDO_REALIZADO",
        "CANTIDAD_ENDOSOS_RECIBIDOS",
        "CANTIDAD_ENDOSOS_REALIZADOS",
        "CANTIDAD_RUC_EMISORES_RELACIONADOS",
        "CANTIDAD_RUC_BENEFICIARIOS_RELACIONADOS",
        "RATIO_SALIDA_ENTRADA"
    },

    Estadisticas =
        List.Transform(
            Indicadores,
            (Indicador) =>
                let
                    Valores =
                        List.Sort(
                            List.RemoveNulls(
                                Table.Column(Origen, Indicador)
                            ),
                            Order.Ascending
                        ),

                    Cantidad = List.Count(Valores),

                    Percentil =
                        (P as number) as nullable number =>
                            if Cantidad = 0 then
                                null
                            else
                                let
                                    Posicion = (Cantidad - 1) * P,
                                    Inferior = Number.RoundDown(Posicion),
                                    Superior = Number.RoundUp(Posicion),
                                    Fraccion = Posicion - Inferior,
                                    ValorInferior = Valores{Inferior},
                                    ValorSuperior = Valores{Superior}
                                in
                                    ValorInferior +
                                    (ValorSuperior - ValorInferior) * Fraccion
                in
                    [
                        INDICADOR = Indicador,
                        CANTIDAD = Cantidad,
                        MINIMO = if Cantidad = 0 then null else List.Min(Valores),
                        P25 = Percentil(0.25),
                        MEDIANA = Percentil(0.50),
                        P75 = Percentil(0.75),
                        P90 = Percentil(0.90),
                        P95 = Percentil(0.95),
                        MAXIMO = if Cantidad = 0 then null else List.Max(Valores),
                        PROMEDIO = if Cantidad = 0 then null else List.Average(Valores)
                    ]
        ),

    Resultado =
        Table.FromRecords(Estadisticas),

    Tipos =
        Table.TransformColumnTypes(
            Resultado,
            {
                {"INDICADOR", type text},
                {"CANTIDAD", Int64.Type},
                {"MINIMO", type number},
                {"P25", type number},
                {"MEDIANA", type number},
                {"P75", type number},
                {"P90", type number},
                {"P95", type number},
                {"MAXIMO", type number},
                {"PROMEDIO", type number}
            }
        )
in
    Tipos
