qry24_PercentilesIndicadores


let
    Origen =
        Table.Buffer(qry22_ComparacionEntidades),

    Indicadores = {
        "TOTAL_RECIBIDO",
        "TOTAL_REALIZADO",
        "CANTIDAD_ENDOSOS_RECIBIDOS",
        "CANTIDAD_RUC_EMISORES_RELACIONADOS",
        "CANTIDAD_RUC_BENEFICIARIOS_RELACIONADOS"
    },

    CalcularPercentil =
        (Valores as list, P as number) as number =>
            let
                Cantidad = List.Count(Valores),
                Posicion = (Cantidad - 1) * P,
                Inferior = Number.RoundDown(Posicion),
                Superior = Number.RoundUp(Posicion),
                Fraccion = Posicion - Inferior,
                ValorInferior = Valores{Inferior},
                ValorSuperior = Valores{Superior}
            in
                ValorInferior +
                (ValorSuperior - ValorInferior) * Fraccion,

    CalcularEstadisticas =
        (Nombre as text) as record =>
            let
                Valores =
                    List.Sort(
                        List.RemoveNulls(
                            Table.Column(Origen, Nombre)
                        ),
                        Order.Ascending
                    ),

                Cantidad = List.Count(Valores),

                Resultado = [
                    INDICADOR = Nombre,
                    CANTIDAD_ENTIDADES = Cantidad,
                    P75 = CalcularPercentil(Valores, 0.75),
                    P90 = CalcularPercentil(Valores, 0.90),
                    P95 = CalcularPercentil(Valores, 0.95)
                ]
            in
                Resultado,

    Registros =
        List.Transform(
            Indicadores,
            each CalcularEstadisticas(_)
        ),

    Resultado =
        Table.FromRecords(Registros),

    Tipos =
        Table.TransformColumnTypes(
            Resultado,
            {
                {"INDICADOR", type text},
                {"CANTIDAD_ENTIDADES", Int64.Type},
                {"P75", type number},
                {"P90", type number},
                {"P95", type number}
            }
        )
in
    Tipos
