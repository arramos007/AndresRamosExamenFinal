let
    Origen = qryAristasRed,

    FiltrarEndosos = Table.SelectRows(
        Origen,
        each
            [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO"
            or [CLASIFICACION_MOVIMIENTO] = "ENDOSO_RECIBIDO"
    ),

    AgruparMovimientos = Table.Group(
        FiltrarEndosos,
        {"CODIGO MOV."},
        {
            {
                "TOTAL_REGISTROS",
                each Table.RowCount(_),
                Int64.Type
            },
            {
                "EMISORES",
                each List.Count(
                    List.Distinct(
                        Table.Column(
                            _,
                            "IDENTIFICACION DEL EMISOR / ENDOSANTE"
                        )
                    )
                ),
                Int64.Type
            },
            {
                "BENEFICIARIOS",
                each List.Count(
                    List.Distinct(
                        Table.Column(
                            _,
                            "IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION"
                        )
                    )
                ),
                Int64.Type
            }
        }
    ),

    Ordenar = Table.Sort(
        AgruparMovimientos,
        {{"TOTAL_REGISTROS", Order.Descending}}
    )
in
    Ordenar
