qry48_RutasCompletas


let
    // Fuentes y función de expansión
    Indice =
        qry34_IndiceConexionesPorParticipante,

    EstadoInicial =
        qry38_EstadoInicialRutas,

    // NIVEL 1
    Nivel1Lista =
        fnExpandirUnNivel(EstadoInicial, Indice),

    Nivel1 =
        if List.IsEmpty(Nivel1Lista) then
            #table({}, {})
        else
            Table.FromRecords(Nivel1Lista),

    Nivel1Identificado =
        Table.AddColumn(
            Nivel1,
            "NIVEL_EXPANSION",
            each 1,
            Int64.Type
        ),

    AbiertasNivel1 =
        Table.SelectRows(
            Nivel1,
            each [ESTADO_RUTA] = "ABIERTA"
        ),

    // NIVEL 2
    Nivel2Lista =
        fnExpandirUnNivel(AbiertasNivel1, Indice),

    Nivel2 =
        if List.IsEmpty(Nivel2Lista) then
            #table({}, {})
        else
            Table.FromRecords(Nivel2Lista),

    Nivel2Identificado =
        Table.AddColumn(
            Nivel2,
            "NIVEL_EXPANSION",
            each 2,
            Int64.Type
        ),

    AbiertasNivel2 =
        Table.SelectRows(
            Nivel2,
            each [ESTADO_RUTA] = "ABIERTA"
        ),

    // NIVEL 3
    Nivel3Lista =
        fnExpandirUnNivel(AbiertasNivel2, Indice),

    Nivel3 =
        if List.IsEmpty(Nivel3Lista) then
            #table({}, {})
        else
            Table.FromRecords(Nivel3Lista),

    Nivel3Identificado =
        Table.AddColumn(
            Nivel3,
            "NIVEL_EXPANSION",
            each 3,
            Int64.Type
        ),

    AbiertasNivel3 =
        Table.SelectRows(
            Nivel3,
            each [ESTADO_RUTA] = "ABIERTA"
        ),

    // NIVEL 4
    Nivel4Lista =
        fnExpandirUnNivel(AbiertasNivel3, Indice),

    Nivel4 =
        if List.IsEmpty(Nivel4Lista) then
            #table({}, {})
        else
            Table.FromRecords(Nivel4Lista),

    Nivel4Identificado =
        Table.AddColumn(
            Nivel4,
            "NIVEL_EXPANSION",
            each 4,
            Int64.Type
        ),

    AbiertasNivel4 =
        Table.SelectRows(
            Nivel4,
            each [ESTADO_RUTA] = "ABIERTA"
        ),

    // NIVEL 5
    Nivel5Lista =
        fnExpandirUnNivel(AbiertasNivel4, Indice),

    Nivel5 =
        if List.IsEmpty(Nivel5Lista) then
            #table({}, {})
        else
            Table.FromRecords(Nivel5Lista),

    Nivel5Identificado =
        Table.AddColumn(
            Nivel5,
            "NIVEL_EXPANSION",
            each 5,
            Int64.Type
        ),

    // CONSOLIDAR LOS CINCO NIVELES
    ConsolidarNiveles =
        Table.Combine(
            {
                Nivel1Identificado,
                Nivel2Identificado,
                Nivel3Identificado,
                Nivel4Identificado,
                Nivel5Identificado
            }
        ),

    // ELIMINAR SOLAMENTE IDENTIFICADORES DE RUTA REPETIDOS
    RutasUnicas =
        Table.Distinct(
            ConsolidarNiveles,
            {"ID_RUTA_ACTUAL"}
        ),

    // ORDENAR POR PROFUNDIDAD Y RUTA
    OrdenarRutas =
        Table.Sort(
            RutasUnicas,
            {
                {"NIVEL_EXPANSION", Order.Ascending},
                {"ID_RUTA_ACTUAL", Order.Ascending}
            }
        )
in
    OrdenarRutas
