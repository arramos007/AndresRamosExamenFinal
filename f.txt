let
    Origen = qryAuditoriaDuplicadosExactos,

    Agrupar = Table.Group(
        Origen,
        {
            "IDENTIFICACION EMPLEADO (EF)",
            "CODIGO MOV."
        },
        {
            {
                "EMPRESAS_RELACIONADAS",
                each List.Count(
                    List.Distinct(
                        List.RemoveNulls(
                            Table.Column(_, "RUC SOCIAL EMPLEADOR")
                        )
                    )
                ),
                Int64.Type
            },
            {
                "REGISTROS_EXCEDENTES",
                each List.Max(
                    List.RemoveNulls(
                        Table.Column(_, "REGISTROS_EXCEDENTES")
                    )
                ),
                Int64.Type
            },
            {
                "MONTO_EXCEDENTE_CANDIDATO",
                each List.Max(
                    List.RemoveNulls(
                        Table.Column(_, "MONTO_EXCEDENTE_CANDIDATO")
                    )
                ),
                type number
            }
        }
    ),

    Ordenar = Table.Sort(
        Agrupar,
        {
            {"MONTO_EXCEDENTE_CANDIDATO", Order.Descending}
        }
    )
in
    Ordenar
