qry41_ValidacionExpansionNivel1


let
    Origen = qry40_PruebaExpansionNivel1,

    TotalNuevasRutas =
        Table.RowCount(Origen),

    SaltosCorrectos =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [NUMERO_SALTO] = 2
            )
        ),

    RutasConPadre =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ID_RUTA_PADRE] <> null
            )
        ),

    HistorialParticipantesCorrecto =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each
                    List.Count([PARTICIPANTES_VISITADOS]) = 3
            )
        ),

    HistorialOperacionesCorrecto =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each
                    List.Count([OPERACIONES_RECORRIDAS]) = 2
            )
        ),

    ImportesIguales =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each
                    Number.Abs(
                        [MONTO_DEBITO] - [MONTO_CREDITO]
                    ) < 0.000001
            )
        ),

    CiclosDetectados =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ES_CICLO] = true
            )
        ),

    Validaciones = {
        {"TOTAL_NUEVAS_RUTAS", TotalNuevasRutas, TotalNuevasRutas},
        {"NUMERO_SALTO_CORRECTO", SaltosCorrectos, TotalNuevasRutas},
        {"RUTAS_CON_PADRE", RutasConPadre, TotalNuevasRutas},
        {"HISTORIAL_PARTICIPANTES", HistorialParticipantesCorrecto, TotalNuevasRutas},
        {"HISTORIAL_OPERACIONES", HistorialOperacionesCorrecto, TotalNuevasRutas},
        {"IMPORTES_IGUALES", ImportesIguales, TotalNuevasRutas},
        {"CICLOS_DETECTADOS", CiclosDetectados, null}
    },

    Tabla =
        Table.FromRows(
            Validaciones,
            {"VALIDACION", "RESULTADO", "ESPERADO"}
        ),

    AgregarEstado =
        Table.AddColumn(
            Tabla,
            "ESTADO",
            each
                if [ESPERADO] = null
                    or [RESULTADO] = [ESPERADO]
                then "CORRECTO"
                else "REVISAR",
            type text
        )
in
    AgregarEstado
