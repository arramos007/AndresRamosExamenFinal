qry47_ControlCrecimientoQuintoNivel


let
    Indice = qry34_IndiceConexionesPorParticipante,
    EstadoInicial = qry38_EstadoInicialRutas,

    // NIVEL 1
    L1Lista = fnExpandirUnNivel(EstadoInicial, Indice),
    L1 = Table.FromRecords(L1Lista),
    A1 = Table.SelectRows(L1, each [ESTADO_RUTA] = "ABIERTA"),

    // NIVEL 2
    L2Lista = fnExpandirUnNivel(A1, Indice),
    L2 = Table.FromRecords(L2Lista),
    A2 = Table.SelectRows(L2, each [ESTADO_RUTA] = "ABIERTA"),

    // NIVEL 3
    L3Lista = fnExpandirUnNivel(A2, Indice),
    L3 = Table.FromRecords(L3Lista),
    A3 = Table.SelectRows(L3, each [ESTADO_RUTA] = "ABIERTA"),

    // NIVEL 4
    L4Lista = fnExpandirUnNivel(A3, Indice),
    L4 = Table.FromRecords(L4Lista),
    A4 = Table.SelectRows(L4, each [ESTADO_RUTA] = "ABIERTA"),

    // NIVEL 5
    L5Lista = fnExpandirUnNivel(A4, Indice),

    L5 =
        if List.IsEmpty(L5Lista) then
            #table(
                {
                    "ESTADO_RUTA",
                    "ID_REGISTRO_DEBITO",
                    "ID_PARTICIPANTE_INICIAL",
                    "ID_PARTICIPANTE_ACTUAL",
                    "ES_CICLO"
                },
                {}
            )
        else
            Table.FromRecords(L5Lista),

    Agrupar =
        Table.Group(
            L5,
            {"ESTADO_RUTA"},
            {
                {
                    "CANTIDAD_RUTAS",
                    each Table.RowCount(_),
                    Int64.Type
                },
                {
                    "OPERACIONES_DISTINTAS",
                    each List.Count(
                        List.Distinct([ID_REGISTRO_DEBITO])
                    ),
                    Int64.Type
                },
                {
                    "PARTICIPANTES_INICIALES",
                    each List.Count(
                        List.Distinct([ID_PARTICIPANTE_INICIAL])
                    ),
                    Int64.Type
                },
                {
                    "PARTICIPANTES_DESTINO",
                    each List.Count(
                        List.Distinct([ID_PARTICIPANTE_ACTUAL])
                    ),
                    Int64.Type
                }
            }
        ),

    TotalRutas = Table.RowCount(L5),

    TotalOperaciones =
        if TotalRutas = 0 then 0
        else List.Count(
            List.Distinct(L5[ID_REGISTRO_DEBITO])
        ),

    TotalCiclos =
        if TotalRutas = 0 then 0
        else Table.RowCount(
            Table.SelectRows(L5, each [ES_CICLO] = true)
        ),

    Resumen =
        #table(
            {
                "ESTADO_RUTA",
                "CANTIDAD_RUTAS",
                "OPERACIONES_DISTINTAS",
                "PARTICIPANTES_INICIALES",
                "PARTICIPANTES_DESTINO"
            },
            {
                {
                    "TOTAL_GENERAL",
                    TotalRutas,
                    TotalOperaciones,
                    null,
                    null
                },
                {
                    "TOTAL_CICLOS",
                    TotalCiclos,
                    null,
                    null,
                    null
                }
            }
        ),

    Resultado = Table.Combine({Agrupar, Resumen})
in
    Resultado
