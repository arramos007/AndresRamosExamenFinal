let
    Origen = qryAuditoriaDuplicadosExactos,

    Filtrar = Table.SelectRows(
        Origen,
        each
            [#"IDENTIFICACION EMPLEADO (EF)"] = "0926877341001"
            and [#"CODIGO MOV."] = 45493225
    ),

    Seleccionar = Table.SelectColumns(
        Filtrar,
        {
            "IDENTIFICACION EMPLEADO (EF)",
            "RUC SOCIAL EMPLEADOR",
            "CODIGO MOV.",
            "CODIGO LIQUID.",
            "ACTO ADMINIS.",
            "FECHA DE MOV.",
            "TOTAL_REPETICIONES",
            "REGISTROS_EXCEDENTES",
            "MONTO MOV.",
            "MONTO_EXCEDENTE_CANDIDATO"
        }
    ),

    Ordenar = Table.Sort(
        Seleccionar,
        {
            {"RUC SOCIAL EMPLEADOR", Order.Ascending}
        }
    )
in
    Ordenar
