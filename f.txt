qry08_EndososDebito


let
    Origen = qry06_MovimientosConParticipantes,

    FiltrarEndososDebito =
        Table.SelectRows(
            Origen,
            each [CLASIFICACION_MOVIMIENTO] = "ENDOSO_REALIZADO"
        ),

    SeleccionarColumnas =
        Table.SelectColumns(
            FiltrarEndososDebito,
            {
                "ID_REGISTRO_ORIGEN",
                "ID_PARTICIPANTE_TITULAR",
                "ID_PARTICIPANTE_EMISOR",
                "ID_PARTICIPANTE_BENEFICIARIO",
                "IDENTIFICACION EMPLEADO (EF)",
                "NOMBRE EMPLEADO (EF)",
                "IDENTIFICACION DEL EMISOR / ENDOSANTE",
                "NOMBRE DEL EMISOR / ENDOSANTE",
                "IDENTIFICACION DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION",
                "NOMBRE DEL BENEFICIARIO / EMISION / ENDOSO / UTILIZACION",
                "FECHA DE MOV.",
                "MONTO MOV.",
                "CODIGO MOV.",
                "CODIGO LIQUID.",
                "INST.",
                "SALDO NCD",
                "EN TRAMA EMISOR",
                "EN TRAMA BENEF."
            }
        ),

    Resultado =
        Table.Sort(
            SeleccionarColumnas,
            {
                {"FECHA DE MOV.", Order.Ascending},
                {"ID_REGISTRO_ORIGEN", Order.Ascending}
            }
        )
in
    Resultado
