qry39_ValidacionEstadoInicial


let
    Origen = qry38_EstadoInicialRutas,

    TotalRutas =
        Table.RowCount(Origen),

    DebitosDistintos =
        List.Count(
            List.Distinct(Origen[ID_REGISTRO_DEBITO])
        ),

    SaltosIniciales =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [NUMERO_SALTO] = 1
            )
        ),

    ParticipantesPorRuta =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each List.Count([PARTICIPANTES_VISITADOS]) = 2
            )
        ),

    OperacionesPorRuta =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each List.Count([OPERACIONES_RECORRIDAS]) = 1
            )
        ),

    RutasSinPadre =
        Table.RowCount(
            Table.SelectRows(
                Origen,
                each [ID_RUTA_PADRE] = null
            )
        ),

    Validaciones = {
        {"TOTAL_RUTAS_INICIALES", TotalRutas, 1338},
        {"DEBITOS_DISTINTOS", DebitosDistintos, 1338},
        {"NUMERO_SALTO_IGUAL_A_UNO", SaltosIniciales, 1338},
        {"DOS_PARTICIPANTES_POR_RUTA", ParticipantesPorRuta, 1338},
        {"UNA_OPERACION_POR_RUTA", OperacionesPorRuta, 1338},
        {"RUTAS_SIN_PADRE", RutasSinPadre, 1338}
    },

    Tabla =
        Table.FromRows(
            Validaciones,
            {"VALIDACION", "RESULTADO", "ESPERADO"}
        ),

    AgregarEstado =
        Table.AddColumn(
            Tabla,
            "ESTADO",
            each
                if [RESULTADO] = [ESPERADO] then
                    "CORRECTO"
                else
                    "REVISAR",
            type text
        )
in
    AgregarEstado
