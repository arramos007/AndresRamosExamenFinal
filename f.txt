qry40_PruebaExpansionNivel1


let
    // Tomar una muestra de 10 rutas iniciales
    EstadoPrueba =
        Table.FirstN(
            qry38_EstadoInicialRutas,
            10
        ),

    // Ejecutar la expansión de un nivel
    NuevasRutas =
        fnExpandirUnNivel(
            EstadoPrueba,
            qry34_IndiceConexionesPorParticipante
        ),

    // Convertir el resultado en tabla
    Resultado =
        if List.IsEmpty(NuevasRutas) then
            #table(
                {"RESULTADO"},
                {{"SIN_CONEXIONES_POSTERIORES"}}
            )
        else
            Table.FromRecords(NuevasRutas),

    // Ordenar por ruta y salto
    Ordenar =
        if List.IsEmpty(NuevasRutas) then
            Resultado
        else
            Table.Sort(
                Resultado,
                {
                    {"ID_RUTA_ACTUAL", Order.Ascending},
                    {"NUMERO_SALTO", Order.Ascending}
                }
            )
in
    Ordenar
