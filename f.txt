let
    Origen = qryDetalleUtilizaciones,

    AgruparPorEmpresa = Table.Group(
        Origen,
        {
            "CLAVE_MOVIMIENTO",
            "RUC SOCIAL EMPLEADOR",
            "ID_TITULAR"
        },
        {
            {
                "REGISTROS_EN_EMPRESA",
                each Table.RowCount(_),
                Int64.Type
            },
            {
                "MONTO_REGISTRADO",
                each List.First(
                    List.RemoveNulls(
                        Table.Column(_, "MONTO MOV.")
                    ),
                    null
                ),
                type number
            },
            {
                "FECHA_PRIMER_REGISTRO",
                each List.Min(
                    Table.Column(_, "FECHA DE MOV.")
                ),
                type datetime
            },
            {
                "FECHA_ULTIMO_REGISTRO",
                each List.Max(
                    Table.Column(_, "FECHA DE MOV.")
                ),
                type datetime
            }
        }
    ),

    FiltrarRepeticionesInternas = Table.SelectRows(
        AgruparPorEmpresa,
        each [REGISTROS_EN_EMPRESA] > 1
    ),

    OrdenarResultados = Table.Sort(
        FiltrarRepeticionesInternas,
        {
            {"REGISTROS_EN_EMPRESA", Order.Descending}
        }
    )
in
    OrdenarResultados
