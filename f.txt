qry26_ResumenPerfilesEntidades


let
    Origen =
        Table.Buffer(qry25_PerfilesPreliminaresEntidades),

    TotalEntidades =
        Table.RowCount(Origen),

    Resumen = {
        [
            PERFIL = "POSIBLE_CONCENTRADORA",
            CANTIDAD =
                Table.RowCount(
                    Table.SelectRows(
                        Origen,
                        each [POSIBLE_CONCENTRADORA] = true
                    )
                )
        ],
        [
            PERFIL = "POSIBLE_ACUMULADORA",
            CANTIDAD =
                Table.RowCount(
                    Table.SelectRows(
                        Origen,
                        each [POSIBLE_ACUMULADORA] = true
                    )
                )
        ],
        [
            PERFIL = "POSIBLE_INTERMEDIARIA",
            CANTIDAD =
                Table.RowCount(
                    Table.SelectRows(
                        Origen,
                        each [POSIBLE_INTERMEDIARIA] = true
                    )
                )
        ],
        [
            PERFIL = "POSIBLE_REDISTRIBUIDORA",
            CANTIDAD =
                Table.RowCount(
                    Table.SelectRows(
                        Origen,
                        each [POSIBLE_REDISTRIBUIDORA] = true
                    )
                )
        ],
        [
            PERFIL = "DOS_O_MAS_PERFILES",
            CANTIDAD =
                Table.RowCount(
                    Table.SelectRows(
                        Origen,
                        each [CANTIDAD_PERFILES] >= 2
                    )
                )
        ],
        [
            PERFIL = "SIN_PERFIL_PRELIMINAR",
            CANTIDAD =
                Table.RowCount(
                    Table.SelectRows(
                        Origen,
                        each [CANTIDAD_PERFILES] = 0
                    )
                )
        ]
    },

    Resultado =
        Table.FromRecords(Resumen),

    AgregarTotal =
        Table.AddColumn(
            Resultado,
            "TOTAL_ENTIDADES",
            each TotalEntidades,
            Int64.Type
        ),

    AgregarPorcentaje =
        Table.AddColumn(
            AgregarTotal,
            "PORCENTAJE",
            each
                if [TOTAL_ENTIDADES] = 0 then
                    0
                else
                    [CANTIDAD] / [TOTAL_ENTIDADES] * 100,
            type number
        )
in
    AgregarPorcentaje
